    Chapter 14 – Solutions¶

    Solutions of the Lab and the Exercises of Chapter 14. Try each task before reading its solution. The code is in the chapter's src folder: ybus_check.m (the worked example, Octave + MATPOWER) and ybus_numpy.py (the same in NumPy against pandapower, Lab steps 1–4), small_ybus.py (Exercises 1–2), bdc_check.m (Exercise 3, Octave), kron_reduction.py (Exercise 4) and long_line.py (Exercise 5). Each Python module's main() prints every number quoted below; the Octave numbers come from GNU Octave 11.3 with MATPOWER 8.1 (pandapower 3.5 for the Python path). Where the two paths overlap they agree to round-off.

    Lab¶

    1. Ybus from the tables. ybus_numpy.case14_ppc() loads pandapower.networks.case14(), runs one power flow and returns net._ppc: 14 buses, 20 branches, baseMVA = 100. connection_matrices builds \(C_f\) and \(C_t\) (20 × 14, one 1 per row, from the F_BUS and T_BUS columns, already 0-based in the ppc), branch_admittances the four entries of every branch from BR_R, BR_X, BR_B, TAP and SHIFT,

      \[ y_{tt} = y_s + j\tfrac{b_c}{2}, \quad y_{ff} = \frac{y_{tt}}{\tau^2}, \quad y_{ft} = -\frac{y_s}{\tau e^{-j\theta}}, \quad y_{tf} = -\frac{y_s}{\tau e^{j\theta}}, \]

      and build_ybus assembles \(Y_f = [Y_{ff}] C_f + [Y_{ft}] C_t\), \(Y_t = [Y_{tf}] C_f + [Y_{tt}] C_t\) and \(Y_\text{bus} = C_f^\top Y_f + C_t^\top Y_t + [Y_\text{sh}]\) with \(Y_\text{sh} = (G_s + jB_s)/\text{baseMVA}\) from the bus table (only bus 9 has a shunt, 19 Mvar). The Octave path is ybus_check.m unchanged.

    2. Comparison. max_difference(ppc) gives \(\max |Y_\text{bus} - Y_\text{bus}^\text{makeYbus}| = 0\) exactly: both codes do the same floating-point operations in the same order. Octave: ybus_check('case14') returns \(1.8 \times 10^{-15}\) and ybus_check('case57') \(1.4 \times 10^{-14}\), round-off from the sparse products.

    3. Sparsity, taps, symmetry. matplotlib.pyplot.spy(Ybus) shows 54 nonzeros of 196 (28 %): the 14 diagonal entries and two per branch (figure described, not saved; with 20 branches and 14 buses the pattern is the adjacency of the IEEE 14 graph plus the diagonal). The transformers are the branches with TAP different from 1: branches 15, 16, 17 of the ppc (0-based), i.e. 4–7 with \(\tau = 0.978\), 4–9 with \(\tau = 0.969\) and 5–6 with \(\tau = 0.932\) (1-based bus numbers), none with a phase shift. \(\lVert Y_\text{bus} - Y_\text{bus}^\top \rVert_F = 0\): an off-nominal tap scales \(y_{ff}\) by \(1/\tau^2\) and both off-diagonals by \(1/\tau\), so the matrix stays symmetric; only a phase shift breaks the symmetry.

    4. A 5° phase shift. with_shift(branch, 15, 5.0) puts \(\theta = 5°\) on the 4–7 transformer. Rebuilt, \(\lVert Y_\text{bus} - Y_\text{bus}^\top \rVert_F = 1.2053\), coming from the single pair \(Y_{4,7} = -0.4261 + 4.8709j\) and \(Y_{7,4} = +0.4261 + 4.8709j\): the two entries have the same modulus 4.8895 (\(= 1/(x\tau)\)) and opposite real parts, since \(y_{ft} = -y_s e^{j\theta}/\tau\) and \(y_{tf} = -y_s e^{-j\theta}/\tau\) are complex conjugates up to the common factor \(-y_s/\tau\). The norm is \(\sqrt{2}\,|Y_{4,7} - Y_{7,4}| = \sqrt{2} \times 2 \times 0.4261 = 1.2053\), i.e. \(2\sqrt{2}\,\sin 5° / (x \tau)\).

      ybus_numpy.py
      """Ybus of IEEE 14 from the connection matrices, checked against pandapower's makeYbus (Chapter 14, Lab).

Every branch is a pi section with series admittance ys = 1/(r + jx), total charging bc and an ideal transformer of
complex ratio tap = tau·exp(j·shift) at the from end. Its four entries are Yff = (ys + j·bc/2)/|tap|²,
Yft = −ys/conj(tap), Ytf = −ys/tap and Ytt = ys + j·bc/2. With Cf and Ct marking the from and to bus of each
branch, Yf = diag(Yff) Cf + diag(Yft) Ct, Yt = diag(Ytf) Cf + diag(Ytt) Ct and Ybus = Cfᵀ Yf + Ctᵀ Yt + diag(Ysh).
pandapower keeps a MATPOWER-style ``ppc`` after a power flow; it stores TAP = 1 for lines where MATPOWER files
write 0, so the tap column can be used as it is.
"""

import numpy as np
import pandapower as pp
import pandapower.networks as pn
from pandapower.pypower.idx_brch import BR_B, BR_R, BR_STATUS, BR_X, F_BUS, SHIFT, T_BUS, TAP
from pandapower.pypower.idx_bus import BS, GS
from pandapower.pypower.makeYbus import makeYbus


def case14_ppc() -> dict:
    """The internal MATPOWER-style case of pandapower's IEEE 14, populated by one power flow."""
    net = pn.case14()
    pp.runpp(net)
    return net._ppc  # noqa: SLF001  (pandapower exposes the ppc only as a private attribute)


def connection_matrices(branch: np.ndarray, n_bus: int) -> tuple[np.ndarray, np.ndarray]:
    """Cf and Ct, shape (branches, buses): a one in the column of the from bus and of the to bus of each branch."""
    n_branch = branch.shape[0]
    cf, ct = np.zeros((n_branch, n_bus)), np.zeros((n_branch, n_bus))
    cf[np.arange(n_branch), branch[:, F_BUS].real.astype(int)] = 1
    ct[np.arange(n_branch), branch[:, T_BUS].real.astype(int)] = 1
    return cf, ct


def branch_admittances(branch: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """The four entries Yff, Yft, Ytf, Ytt of every branch, out-of-service branches set to zero."""
    status = branch[:, BR_STATUS].real
    ys = status / (branch[:, BR_R] + 1j * branch[:, BR_X])
    bc = status * branch[:, BR_B].real
    tau = np.where(branch[:, TAP].real == 0, 1.0, branch[:, TAP].real)
    tap = tau * np.exp(1j * np.pi / 180 * branch[:, SHIFT].real)
    ytt = ys + 1j * bc / 2
    yff = ytt / (tap * np.conj(tap))
    yft = -ys / np.conj(tap)
    ytf = -ys / tap
    return yff, yft, ytf, ytt


def build_ybus(bus: np.ndarray, branch: np.ndarray, base_mva: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Ybus, Yf and Yt (dense) assembled as Ybus = Cfᵀ Yf + Ctᵀ Yt + diag(Ysh)."""
    n_bus = bus.shape[0]
    cf, ct = connection_matrices(branch, n_bus)
    yff, yft, ytf, ytt = branch_admittances(branch)
    yf = np.diag(yff) @ cf + np.diag(yft) @ ct
    yt = np.diag(ytf) @ cf + np.diag(ytt) @ ct
    ysh = (bus[:, GS] + 1j * bus[:, BS]) / base_mva
    return cf.T @ yf + ct.T @ yt + np.diag(ysh), yf, yt


def max_difference(ppc: dict) -> float:
    """Largest absolute entry of Ybus (this module) minus Ybus (pandapower's makeYbus)."""
    ybus, _, _ = build_ybus(ppc['bus'], ppc['branch'], ppc['baseMVA'])
    reference, _, _ = makeYbus(ppc['baseMVA'], ppc['bus'], ppc['branch'])
    return float(np.abs(ybus - reference.toarray()).max())


def transformer_branches(branch: np.ndarray) -> np.ndarray:
    """Indices of the branches with an off-nominal tap or a phase shift."""
    tap, shift = branch[:, TAP].real, branch[:, SHIFT].real
    return np.flatnonzero(((tap != 0) & (tap != 1)) | (shift != 0))


def asymmetry(matrix: np.ndarray) -> float:
    """Frobenius norm of Y − Yᵀ, zero for a symmetric matrix."""
    return float(np.linalg.norm(matrix - matrix.T))


def with_shift(branch: np.ndarray, index: int, degrees: float) -> np.ndarray:
    """A copy of the branch table with a phase shift (degrees) on one branch."""
    shifted = branch.copy()
    shifted[index, SHIFT] = degrees
    return shifted


def main() -> None:
    """Print the numbers of Lab steps 1–4 for IEEE 14."""
    ppc = case14_ppc()
    bus, branch, base_mva = ppc['bus'], ppc['branch'], ppc['baseMVA']
    ybus, _, _ = build_ybus(bus, branch, base_mva)
    print(f'{bus.shape[0]} buses, {branch.shape[0]} branches, baseMVA {base_mva}')
    print(f'max |Ybus - makeYbus| = {max_difference(ppc):.3e}')
    nnz = int(np.count_nonzero(ybus))
    print(f'nonzeros {nnz} of {ybus.size} ({100 * nnz / ybus.size:.0f} %)')
    print(f'  {bus.shape[0]} diagonal + 2 x {(nnz - bus.shape[0]) // 2} off-diagonal')
    trafos = transformer_branches(branch)
    for k in trafos:
        f, t = int(branch[k, F_BUS].real), int(branch[k, T_BUS].real)
        tap, x = branch[k, TAP].real, branch[k, BR_X].real
        print(f'transformer branch {k}: buses {f + 1}-{t + 1} (1-based), tap {tap:.4f}, x {x:.5f}')
    print(f'||Ybus - Ybus^T|| = {asymmetry(ybus):.3e} (symmetric despite the taps)')
    shifted, _, _ = build_ybus(bus, with_shift(branch, int(trafos[0]), 5.0), base_mva)
    f, t = int(branch[trafos[0], F_BUS].real), int(branch[trafos[0], T_BUS].real)
    print(f'with a 5 degree shift on branch {trafos[0]}: ||Ybus - Ybus^T|| = {asymmetry(shifted):.4f}')
    print(f'  Y[{f + 1},{t + 1}] = {shifted[f, t]:.4f}, Y[{t + 1},{f + 1}] = {shifted[t, f]:.4f}')
    print(f'  same modulus {abs(shifted[f, t]):.4f}')


if __name__ == '__main__':
    main()

    Exercise 1¶

    Series admittances: \(1/j0.1 = -j10\) (line 1–2), \(1/j0.2 = -j5\) (line 2–3), \(1/j0.25 = -j4\) (line 1–3). Each line adds \(y_s + jb_c/2 = y_s + j0.01\) to the diagonal entry of both its buses and \(-y_s\) to the two off-diagonal entries between them. Bus 1 has lines 1–2 and 1–3, bus 2 has 1–2 and 2–3, bus 3 has 2–3 and 1–3:

    \[ Y_\text{bus} = \begin{bmatrix} -j13.98 & j10 & j4 \\ j10 & -j14.98 & j5 \\ j4 & j5 & -j8.98 \end{bmatrix}, \]

    for example \(Y_{11} = -j10 - j4 + 2 \times j0.01 = -j13.98\). Every row sums to \(j0.02\): the series admittances cancel between the diagonal and the off-diagonal entries of the same row, and what remains is the total charging connected to the bus, two half-lines of \(j0.01\) each. In general \(\sum_k Y_{ik}\) is the total shunt admittance at bus \(i\) (charging plus explicit shunts), zero for a lossless network without charging, which is why \(B_\text{bus}\) of the DC power flow is singular. small_ybus.ring3_ybus() and row_sums reproduce the matrix and the row sums of \(j0.02\).

    Exercise 2¶

    With \(y_s = 1/(j0.1) = -j10\), \(\tau = 1.05\) and \(\theta = 0\),

    \[ Y_\text{br} = \begin{bmatrix} y_s/\tau^2 & -y_s/\tau \\ -y_s/\tau & y_s \end{bmatrix} = \begin{bmatrix} -j9.0703 & j9.5238 \\ j9.5238 & -j10 \end{bmatrix}. \]

    A \(\pi\) circuit with series admittance \(y_\text{se}\) and shunts \(y_1\), \(y_2\) has \(Y = \begin{bmatrix} y_\text{se} + y_1 & -y_\text{se} \\ -y_\text{se} & y_\text{se} + y_2 \end{bmatrix}\), so matching the entries:

    \[ y_\text{se} = \frac{y_s}{\tau} = -j9.5238, \qquad y_1 = y_s\left(\frac{1}{\tau^2} - \frac{1}{\tau}\right) = +j0.4535, \qquad y_2 = y_s\left(1 - \frac{1}{\tau}\right) = -j0.4762 . \]

    The shunt at the tapped (from) side is capacitive (\(+j\)) and the one at the other side inductive (\(-j\)); they nearly cancel (\(y_1 + y_2 = y_s (1/\tau - 1)^2 = -j0.0227\)). The circuit is asymmetric because the ideal transformer sits at one end: seen from bus 1 the reactance is scaled by \(\tau^2\), seen from bus 2 it is not, so the two driving-point admittances \(y_s/\tau^2\) and \(y_s\) differ while the transfer admittance is the geometric mean \(y_s/\tau\) of the two. A symmetric \(\pi\) would need \(y_1 = y_2\), i.e. \(\tau = 1\). The "negative" shunt is a mathematical artefact of forcing a transformer into a \(\pi\) template, not a physical capacitor; it is nevertheless what every power-flow program stores. small_ybus.transformer_ybr(1.05, 0.1) and pi_equivalent print these values.

    small_ybus.py
    """Ybus of a three-bus ring by inspection and the branch matrix of a tapped transformer (Chapter 14, Exercises 1–2).

By inspection, the diagonal entry of a bus is the sum of the admittances of everything connected to it (series
admittances of its lines plus half the charging of each line plus shunts) and the off-diagonal entry between two
buses is minus the series admittance of the line joining them. For a transformer with ratio tau at the from end,
Ybr = [[ys/tau², −ys/tau], [−ys/tau, ys]], whose pi equivalent has series admittance ys/tau, a from-side shunt of
ys (1/tau² − 1/tau) and a to-side shunt of ys (1 − 1/tau).
"""

import numpy as np

RING_LINES = [(0, 1, 0.1j), (1, 2, 0.2j), (0, 2, 0.25j)]  # (from, to, series impedance in p.u.)
RING_BC = 0.02  # total charging susceptance of each line, p.u.


def ring3_ybus(lines: list[tuple[int, int, complex]] = RING_LINES, bc: float = RING_BC) -> np.ndarray:
    """Ybus of the three-bus ring, filled in by inspection line by line."""
    ybus = np.zeros((3, 3), dtype=complex)
    for f, t, z in lines:
        ys = 1 / z
        ybus[f, f] += ys + 1j * bc / 2
        ybus[t, t] += ys + 1j * bc / 2
        ybus[f, t] -= ys
        ybus[t, f] -= ys
    return ybus


def row_sums(ybus: np.ndarray) -> np.ndarray:
    """The sum of each row: with no shunt elements it is the total charging admittance connected to the bus."""
    return ybus.sum(axis=1)


def transformer_ybr(tau: float, x: float, shift_deg: float = 0.0) -> np.ndarray:
    """Branch admittance matrix of a transformer with ratio tau·exp(j·shift) at the from end and reactance x."""
    ys = 1 / (1j * x)
    tap = tau * np.exp(1j * np.deg2rad(shift_deg))
    return np.array([[ys / (tap * np.conj(tap)), -ys / np.conj(tap)], [-ys / tap, ys]])


def pi_equivalent(ybr: np.ndarray) -> tuple[complex, complex, complex]:
    """Series admittance, from-side shunt and to-side shunt of the pi circuit with the same branch matrix.

    Valid when the two off-diagonal entries are equal (no phase shift): a pi circuit cannot represent a phase
    shifter, whose Ybr is not symmetric.
    """
    series = -ybr[0, 1]
    return series, ybr[0, 0] - series, ybr[1, 1] - series


def main() -> None:
    """Print the ring's Ybus with its row sums and the transformer's Ybr with its pi equivalent."""
    np.set_printoptions(precision=4, suppress=True)
    ybus = ring3_ybus()
    print('three-bus ring Ybus =')
    print(ybus)
    print('row sums =', row_sums(ybus), '  (expected j bc = 0.02j at every bus)')
    ybr = transformer_ybr(1.05, 0.1)
    print('transformer tau = 1.05, x = 0.1: Ybr =')
    print(ybr)
    series, shunt_from, shunt_to = pi_equivalent(ybr)
    print(f'pi equivalent: series {series:.4f}, shunt at from {shunt_from:.4f}, shunt at to {shunt_to:.4f}')


if __name__ == '__main__':
    main()

    Exercise 3¶

    bdc_check.m builds \(A = C_f - C_t\), \(B_\text{plain} = A^\top \operatorname{diag}(1/x)\, A\) and subtracts it from the first output of makeBdc(mpc). For case14 the difference is zero everywhere except at the buses of the three transformers, where MATPOWER uses \(b = 1/(x\tau)\) instead of \(1/x\) (makeBdc: b = b ./ tap):

    branch \(\tau\) \(x\) \(1/x\) \(1/(x\tau)\) difference
    4–7 0.978 0.20912 4.7819 4.8895 0.10757
    4–9 0.969 0.55618 1.7980 1.8555 0.05752
    5–6 0.932 0.25202 3.9679 4.2574 0.28951

    The eleven entries that differ (1-based) are exactly the ones these branches touch: \((5,5)\) and \((6,6)\) by \(+0.28951\), \((5,6)\) and \((6,5)\) by \(-0.28951\); \((4,7)\) and \((7,4)\) by \(-0.10757\), \((7,7)\) by \(+0.10757\); \((4,9)\) and \((9,4)\) by \(-0.05752\), \((9,9)\) by \(+0.05752\); and \((4,4)\) by \(+0.10757 + 0.05752 = 0.16509\), the sum of its two transformers. The largest difference is 0.28951 (the 5–6 transformer, with the smallest \(\tau\)), and for case9, which has no taps, the two matrices are identical (\(0\)). Nothing else differs: phase shifts, if there were any, would not appear in \(B_\text{bus}\) at all but in the injection vectors Pbusinj and Pfinj (the third and fourth outputs; here zero), because in the DC model a shift adds a constant to the angle difference rather than a term to the matrix. The lesson for get_susceptance_matrices in ToOp: \(b = 1/x\) is the DC model of a line, and the tap ratio of a transformer enters the DC susceptance as \(1/(x\tau)\), a correction of a few percent that MATPOWER applies and a plain Laplacian forgets.

    bdc_check.m
    function [maxdiff, rows, cols, diffs] = bdc_check(casename)
  % Compare the plain DC matrix A' diag(1/x) A, which ignores taps, with the first output of MATPOWER's makeBdc,
  % and return the largest difference with the list of entries that differ (Chapter 14, Exercise 3).
  define_constants;
  mpc = ext2int(loadcase(casename));
  Bbus = makeBdc(mpc);

  nb = size(mpc.bus, 1);  nl = size(mpc.branch, 1);
  Cf = sparse(1:nl, mpc.branch(:, F_BUS), 1, nl, nb);
  Ct = sparse(1:nl, mpc.branch(:, T_BUS), 1, nl, nb);
  A = Cf - Ct;
  b = mpc.branch(:, BR_STATUS) ./ mpc.branch(:, BR_X);
  Bplain = A' * spdiags(b, 0, nl, nl) * A;
  D = Bbus - Bplain;
  [rows, cols, diffs] = find(abs(D) > 1e-9 .* ones(size(D)) & D ~= 0);
  diffs = full(D(sub2ind(size(D), rows, cols)));
  maxdiff = full(max(max(abs(D))));
end

    Exercise 4¶

    Partition the nodal equations \(I = Y V\) into kept (\(k\)) and eliminated (\(e\)) buses:

    \[ \begin{bmatrix} I_k \\ 0 \end{bmatrix} = \begin{bmatrix} Y_{kk} & Y_{ke} \\ Y_{ek} & Y_{ee} \end{bmatrix} \begin{bmatrix} V_k \\ V_e \end{bmatrix} \;\Rightarrow\; V_e = -Y_{ee}^{-1} Y_{ek} V_k, \qquad I_k = \left(Y_{kk} - Y_{ke} Y_{ee}^{-1} Y_{ek}\right) V_k . \]

    The Schur complement \(Y_\text{red} = Y_{kk} - Y_{ke} Y_{ee}^{-1} Y_{ek}\) is the Ward-type equivalent: an admittance matrix on the kept buses alone that gives the same \(V_k\) for every injection with \(I_e = 0\). kron_reduction.KEEP keeps buses 1–5 of IEEE 14 (the 132 kV part) and eliminates 6–14. Checks (kron_reduction.main()): with a deterministic pseudo-random complex injection at the kept buses, \(\max |V_k^\text{full} - V_k^\text{reduced}| = 3.5 \times 10^{-13}\), round-off; the reduced matrix is symmetric (the original is). Only buses 4 and 5 are neighbours of the eliminated set (through the transformers 4–7, 4–9 and 5–6), so the reduction changes only the block \(\{4, 5\}\): \(Y_{4,5}\) goes from \(-6.841 + 21.579j\) to \(-7.101 + 23.060j\), and the diagonals of 4 and 5 move with it, while the entries of buses 1–3 are untouched. In general the elimination creates a fictitious branch between every pair of kept buses that share an eliminated neighbour (fill-in), which is why equivalents of a large external network are dense. The equivalent is only valid for the injection pattern it assumes: adding 1 p.u. of current at the eliminated bus 9 moves the kept voltages by 2.47 p.u. relative to the reduced solution. A Ward equivalent therefore carries, besides \(Y_\text{red}\), equivalent injections at the boundary buses that stand for the eliminated loads and generators, and an extended Ward adds a fictitious generator to hold the boundary voltage.

    kron_reduction.py
    """Kron reduction of the IEEE 14 admittance matrix onto a set of kept buses (Chapter 14, Exercise 4).

Partition the buses into kept (k) and eliminated (e). With zero current injection at the eliminated buses the
nodal equations I = Y V reduce to I_k = (Y_kk − Y_ke Y_ee⁻¹ Y_ek) V_k: the Schur complement is the Ward-type
equivalent, an admittance matrix on the kept buses alone that gives the same kept-bus voltages for any such
injection pattern. Eliminated buses leave fictitious branches between every pair of kept buses they connected, so the
equivalent fills in where the original was sparse.
"""

import numpy as np
from ybus_numpy import build_ybus, case14_ppc

KEEP = np.array([0, 1, 2, 3, 4])  # buses 1–5 (1-based): the 132 kV part of IEEE 14 and the transformer buses


def kron_reduce(ybus: np.ndarray, keep: np.ndarray) -> np.ndarray:
    """Schur complement Y_kk − Y_ke Y_ee⁻¹ Y_ek of the admittance matrix onto the kept buses."""
    eliminate = np.setdiff1d(np.arange(ybus.shape[0]), keep)
    y_kk = ybus[np.ix_(keep, keep)]
    y_ke = ybus[np.ix_(keep, eliminate)]
    y_ek = ybus[np.ix_(eliminate, keep)]
    y_ee = ybus[np.ix_(eliminate, eliminate)]
    return y_kk - y_ke @ np.linalg.solve(y_ee, y_ek)


def kept_voltages_full(ybus: np.ndarray, keep: np.ndarray, injection_kept: np.ndarray) -> np.ndarray:
    """Kept-bus voltages from the full network, with the given currents at the kept buses and zero elsewhere."""
    current = np.zeros(ybus.shape[0], dtype=complex)
    current[keep] = injection_kept
    return np.linalg.solve(ybus, current)[keep]


def kept_voltages_reduced(ybus: np.ndarray, keep: np.ndarray, injection_kept: np.ndarray) -> np.ndarray:
    """Kept-bus voltages from the Kron-reduced matrix alone."""
    return np.linalg.solve(kron_reduce(ybus, keep), injection_kept)


def main() -> None:
    """Print the reduced matrix of IEEE 14 onto buses 1–5 and the voltage check."""
    np.set_printoptions(precision=3, suppress=True, linewidth=119)
    ppc = case14_ppc()
    ybus, _, _ = build_ybus(ppc['bus'], ppc['branch'], ppc['baseMVA'])
    reduced = kron_reduce(ybus, KEEP)
    print(f'kept buses (1-based): {KEEP + 1}, eliminated: {np.setdiff1d(np.arange(14), KEEP) + 1}')
    print(
        'nonzeros: full block Y_kk', np.count_nonzero(ybus[np.ix_(KEEP, KEEP)]), ', reduced', np.count_nonzero(reduced)
    )
    print('reduced matrix (imaginary part):')
    print(reduced.imag)
    print('new coupling between buses 4 and 5 (zero in Y_kk):', f'{ybus[3, 4]:.3f} -> {reduced[3, 4]:.3f}')
    rng = np.random.default_rng(14)
    injection = rng.normal(size=len(KEEP)) + 1j * rng.normal(size=len(KEEP))
    v_full = kept_voltages_full(ybus, KEEP, injection)
    v_reduced = kept_voltages_reduced(ybus, KEEP, injection)
    print(f'max |V_full - V_reduced| at the kept buses = {np.abs(v_full - v_reduced).max():.2e}')
    with_injection_elsewhere = np.zeros(14, dtype=complex)
    with_injection_elsewhere[KEEP] = injection
    with_injection_elsewhere[8] = 1.0  # one ampere at bus 9: the equivalent no longer applies
    v_bad = np.linalg.solve(ybus, with_injection_elsewhere)[KEEP]
    print(f'with a current at an eliminated bus the mismatch is {np.abs(v_bad - v_reduced).max():.3f}')

    The statement leaves the per-km parameters to you; long_line.py uses a typical single-circuit 400 kV line at 50 Hz: \(r = 0.028\ \Omega/\text{km}\), \(L = 0.9\) mH/km (\(x = 0.2827\ \Omega/\text{km}\)), \(C = 12.5\) nF/km (\(b = 3.927\ \mu\text{S/km}\)), \(g = 0\). Then \(\gamma = \sqrt{zy} = 5.21 \times 10^{-5} + j1.055 \times 10^{-3}\) per km (wavelength \(2\pi/\beta = 5956\) km) and \(Z_c = \sqrt{z/y} = 268.7 - j13.3\ \Omega\). For a length \(l\),

    \[ A = D = \cosh \gamma l, \qquad B = Z_c \sinh \gamma l, \qquad C = \frac{\sinh \gamma l}{Z_c}, \qquad AD - BC = 1 , \]

    and at 300 km \(A = 0.9504 + 0.0049j\), \(B = 8.12 + 83.43j\ \Omega\), \(C = -1.9 \times 10^{-6} + 1.159 \times 10^{-3}j\) S. The exact equivalent \(\pi\) has \(Z' = B = Z \,\sinh(\gamma l)/(\gamma l)\) and \(Y'/2 = (A - 1)/B = (Y/2) \tanh(\gamma l/2)/(\gamma l/2)\), against the nominal \(Z = zl = 8.40 + 84.82j\ \Omega\) and \(Y = yl = 1.178 \times 10^{-3}j\) S: the exact series impedance is 1.7 % smaller and the shunt 0.8 % larger. The relative error of the nominal series admittance \(1/Z\) is \(|\gamma l / \sinh(\gamma l) - 1| \approx |\gamma l|^2/6\):

    length (km) 50 100 200 300 400 500
    error of \(1/Z\) (%) 0.05 0.19 0.75 1.69 3.04 4.80

    The plot (described, not saved) is a parabola: the error grows with the square of the length, 1.67 % from the rule of thumb at 300 km against 1.69 % exact, and crosses 1 % at about 230 km. That is the usual textbook boundary between "medium" lines, where the nominal \(\pi\) is adequate, and "long" lines, where the exact \(\pi\) (or the ABCD form) is needed. For the branch tables of Chapter 14 the practical consequence is that the \(r\), \(x\), \(b_c\) of a long line in a MATPOWER or pandapower case should already be the equivalent \(\pi\) values, which is what grid data from TSOs normally contains.

    long_line.py
    """Exact long-line (ABCD) parameters against the nominal pi model as a function of length (Chapter 14, Exercise 5).

For a line with series impedance z and shunt admittance y per km, the propagation constant is gamma = sqrt(z y)
and the characteristic impedance Zc = sqrt(z / y). Over a length l the exact two-port has A = D = cosh(gamma l),
B = Zc sinh(gamma l) and C = sinh(gamma l) / Zc, and the exact equivalent pi has series impedance
Z' = Z sinh(gamma l) / (gamma l) and half-shunts (Y/2) tanh(gamma l / 2) / (gamma l / 2), where Z = z l and
Y = y l are the nominal-pi values. The relative error of the nominal series admittance is
|gamma l / sinh(gamma l) − 1|.
"""

import numpy as np

# a typical single-circuit 400 kV overhead line at 50 Hz
R_PER_KM = 0.028  # ohm/km
X_PER_KM = 2 * np.pi * 50 * 0.9e-3  # ohm/km, from 0.9 mH/km
B_PER_KM = 2 * np.pi * 50 * 12.5e-9  # S/km, from 12.5 nF/km
LENGTH_KM = 300.0


def abcd(length_km: float, r: float = R_PER_KM, x: float = X_PER_KM, b: float = B_PER_KM) -> np.ndarray:
    """The exact ABCD matrix of a lossy line of the given length."""
    z, y = r + 1j * x, 1j * b
    gamma, zc = np.sqrt(z * y), np.sqrt(z / y)
    gl = gamma * length_km
    return np.array([[np.cosh(gl), zc * np.sinh(gl)], [np.sinh(gl) / zc, np.cosh(gl)]])


def nominal_pi(
    length_km: float, r: float = R_PER_KM, x: float = X_PER_KM, b: float = B_PER_KM
) -> tuple[complex, complex]:
    """Series impedance Z = z l and total shunt admittance Y = y l of the nominal pi model."""
    return (r + 1j * x) * length_km, 1j * b * length_km


def exact_pi(
    length_km: float, r: float = R_PER_KM, x: float = X_PER_KM, b: float = B_PER_KM
) -> tuple[complex, complex]:
    """Series impedance and total shunt admittance of the exact equivalent pi, from the ABCD parameters."""
    m = abcd(length_km, r, x, b)
    series = m[0, 1]  # B = Z'
    shunt_half = (m[0, 0] - 1) / m[0, 1]  # (A - 1) / B = Y'/2
    return series, 2 * shunt_half


def series_admittance_error(
    length_km: np.ndarray, r: float = R_PER_KM, x: float = X_PER_KM, b: float = B_PER_KM
) -> np.ndarray:
    """Relative error |Z/Z' − 1| of the nominal series admittance 1/Z against the exact 1/Z', elementwise."""
    gl = np.sqrt((r + 1j * x) * 1j * b) * np.asarray(length_km, dtype=float)
    return np.abs(gl / np.sinh(gl) - 1)


def main() -> None:
    """Print the ABCD parameters at 300 km, the nominal and exact pi values and the error at several lengths."""
    z, y = R_PER_KM + 1j * X_PER_KM, 1j * B_PER_KM
    gamma, zc = np.sqrt(z * y), np.sqrt(z / y)
    print(f'z = {z:.4f} ohm/km, y = {y.imag:.3e}j S/km, Zc = {zc:.1f} ohm, gamma = {gamma:.3e} /km')
    print(f'wavelength 2 pi / Im(gamma) = {2 * np.pi / gamma.imag:.0f} km')
    m = abcd(LENGTH_KM)
    print(f'ABCD at {LENGTH_KM:.0f} km: A = {m[0, 0]:.4f}, B = {m[0, 1]:.2f} ohm, C = {m[1, 0]:.3e} S')
    print(f'  AD - BC = {np.linalg.det(m):.6f}')
    (z_nom, y_nom), (z_ex, y_ex) = nominal_pi(LENGTH_KM), exact_pi(LENGTH_KM)
    print(f'nominal pi: Z = {z_nom:.2f} ohm, Y = {y_nom.imag:.3e}j S')
    print(f'exact pi:   Z = {z_ex:.2f} ohm, Y = {y_ex.imag:.3e}j S')
    error_300 = 100 * series_admittance_error(LENGTH_KM)
    print(f'relative error of the series admittance at {LENGTH_KM:.0f} km: {error_300:.2f} %')
    lengths = np.array([50.0, 100.0, 200.0, 300.0, 400.0, 500.0])
    errors = series_admittance_error(lengths)
    print('length km   ' + ' '.join(f'{v:7.0f}' for v in lengths))
    print('error %     ' + ' '.join(f'{100 * v:7.2f}' for v in errors))
    print(f'quadratic rule of thumb |gamma l|^2 / 6 at 300 km: {100 * abs(gamma * LENGTH_KM) ** 2 / 6:.2f} %')


if __name__ == '__main__':
    main()
