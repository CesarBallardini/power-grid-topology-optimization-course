    Chapter 18 – Solutions¶

    Solutions of the Lab and the Exercises of Chapter 18. Try each task before reading its solution. The code is in the chapter's src folder: ieee14.py (IEEE 14 as a DC grid, with brute-force PTDFs of split, reduced or re-parametrised grids), bsdf.py (the BSDF of the worked example), modf.py (MODF, Woodbury, LODF matrix), compose.py (ToOp's composition order), compose_toop.py (the comparison with ToOp), splits.py, woodbury_timing.py, flops.py and bridges.py. Each module's main() prints the numbers quoted below; the ToOp numbers were produced with ToOp ba1874e on CPU. The notebooks examples.ipynb, lab.ipynb and exercises.ipynb import these modules.

    IEEE 14 comes from pandapower's case14 after rundcpp; the DC model (per-unit reactances with transformer taps, loads and generators, bus 1 as slack) reproduces pandapower's DC flows to \(2 \cdot 10^{-13}\) MW. Branch names are L<from>-<to> for lines and T<from>-<to> for transformers, with 1-based bus numbers; node indices in the code are 0-based (bus 2 is node 1).

    ieee14.py
    """IEEE 14 as a DC model for the labs of Chapter 18: nodes, branches, susceptances and injections from pandapower.

``load_case14()`` reads pandapower's ``case14`` after a DC power flow and returns a ``DcGrid``. The grid can be
split at a node, lose branches or change a susceptance, each time returning a new ``DcGrid`` whose PTDF is
computed from scratch: the brute-force references against which the low-rank updates of the chapter are checked.
Flows are in MW, susceptances in per unit on the system base, bus 0 is the slack.
"""

from __future__ import annotations  # ToOp's interpreter is older than 3.14: methods return the class they live in

from dataclasses import dataclass, replace

import numpy as np
import pandapower as pp
import pandapower.networks as pn


def ptdf(frm: np.ndarray, to: np.ndarray, b: np.ndarray, n_bus: int, slack: int = 0) -> np.ndarray:
    """PTDF (branches x buses) of a DC grid, with zero columns for the slack and for isolated nodes."""
    n_br = len(frm)
    a = np.zeros((n_br, n_bus))
    a[np.arange(n_br), frm], a[np.arange(n_br), to] = 1.0, -1.0
    bf = b[:, None] * a
    bbus = a.T @ bf
    degree = np.bincount(np.concatenate([frm, to]), minlength=n_bus)
    keep = (np.arange(n_bus) != slack) & (degree > 0)  # nodes left without branches (busbar outages) are dropped
    h = np.zeros((n_br, n_bus))
    h[:, keep] = np.linalg.solve(bbus[np.ix_(keep, keep)], bf[:, keep].T).T
    return h


@dataclass(frozen=True)
class DcGrid:
    """A DC grid: from/to nodes, susceptances ``b``, nodal injections ``p`` (MW) and branch names."""

    frm: np.ndarray
    to: np.ndarray
    b: np.ndarray
    p: np.ndarray
    names: list[str]
    slack: int = 0

    @property
    def n_bus(self) -> int:
        """Number of nodes, including nodes added by splits."""
        return len(self.p)

    @property
    def n_branch(self) -> int:
        """Number of branches."""
        return len(self.frm)

    def ptdf(self) -> np.ndarray:
        """PTDF computed from scratch (the reference the updates are compared with)."""
        return ptdf(self.frm, self.to, self.b, self.n_bus, self.slack)

    def flows(self) -> np.ndarray:
        """Branch flows ``H P`` in MW."""
        return self.ptdf() @ self.p

    def branches_at(self, node: int) -> list[int]:
        """Indices of the branches with one end at ``node``."""
        return [k for k in range(self.n_branch) if self.frm[k] == node or self.to[k] == node]

    def split(self, node: int, bus_a: list[int]) -> DcGrid:
        """Physically split ``node``: ``bus_a`` branches stay, the others move to a new node; injections stay."""
        nb = self.n_bus
        frm, to = self.frm.copy(), self.to.copy()
        for k in self.branches_at(node):
            if k not in bus_a:
                if frm[k] == node:
                    frm[k] = nb
                else:
                    to[k] = nb
        return replace(self, frm=frm, to=to, p=np.append(self.p, 0.0))

    def without(self, branches: list[int]) -> DcGrid:
        """The grid with the given branches removed (their PTDF rows are gone, use ``names`` to map back)."""
        keep = [k for k in range(self.n_branch) if k not in branches]
        return replace(self, frm=self.frm[keep], to=self.to[keep], b=self.b[keep], names=[self.names[k] for k in keep])

    def scaled(self, branch: int, factor: float) -> DcGrid:
        """The grid with the susceptance of one branch multiplied by ``factor``."""
        b = self.b.copy()
        b[branch] *= factor
        return replace(self, b=b)

    def moved(self, node_from: int, node_to: int, mw: float) -> DcGrid:
        """The grid with ``mw`` of injection moved from one node to another."""
        p = self.p.copy()
        p[node_from] -= mw
        p[node_to] += mw
        return replace(self, p=p)


def ratings(grid: DcGrid, margin: float = 1.3, floor: float = 40.0) -> np.ndarray:
    """MW limits for a case that ships without any: ``margin`` times the base-case flow, at least ``floor`` MW."""
    return np.maximum(margin * np.abs(grid.flows()), floor)


def from_pandapower(net: pp.pandapowerNet) -> DcGrid:
    """The DC model of a pandapower network with contiguous bus indices and the ``ext_grid`` at bus 0."""
    pp.rundcpp(net)
    branch, bus, gen = net._ppc['branch'], net._ppc['bus'], net._ppc['gen']
    frm, to = branch[:, 0].real.astype(int), branch[:, 1].real.astype(int)
    tap = np.where(branch[:, 8].real == 0, 1.0, branch[:, 8].real)
    b = 1.0 / (branch[:, 3].real * tap)
    p = -bus[:, 2].real.copy()  # loads
    np.add.at(p, gen[:, 0].real.astype(int), gen[:, 1].real)  # generators, slack included
    p[0] -= p.sum()  # the slack balances the losses pandapower assigns to it
    names = [f'L{i + 1}-{j + 1}' for i, j in zip(frm, to, strict=True)]
    for k in range(len(net.line), len(frm)):
        names[k] = 'T' + names[k][1:]
    return DcGrid(frm=frm, to=to, b=b, p=p, names=names)


def load_case14() -> DcGrid:
    """IEEE 14 from pandapower: 20 branches (15 lines, 5 transformers), per-unit reactances from the ppc tables."""
    return from_pandapower(pn.case14())


def main() -> None:
    """Print the base-case flows and check them against pandapower."""
    grid = load_case14()
    net = pn.case14()
    pp.rundcpp(net)
    ref = np.concatenate([net.res_line.p_from_mw.to_numpy(), net.res_trafo.p_hv_mw.to_numpy()])
    f = grid.flows()
    print('largest difference to pandapower rundcpp:', np.abs(f - ref).max(), 'MW')
    for name, flow in zip(grid.names, f, strict=True):
        print(f'{name:7s} {flow:8.2f} MW')


if __name__ == '__main__':
    main()

      splits.py
      """Bus splits on IEEE 14: BSDF against a recomputed PTDF, cross-coupler flows, reassignment and a split scan.

Lab steps 1 and 2 of Chapter 18 (``check_split``, ``reassign``), Exercise 3 (``beta_sum``) and Exercise 4
(``scan_station``: every admissible two-way split of one station, with the cross-coupler flow and the worst
post-split loading).
"""

from itertools import combinations

import numpy as np
from compose import moved_ends, split
from ieee14 import DcGrid, load_case14, ratings


def check_split(grid: DcGrid, station: int, bus_a: list[int]) -> dict[str, float]:
    """Largest differences between the BSDF update and the PTDF of the physically split grid (Lab step 1)."""
    s, p_ext = split(grid.ptdf(), grid.frm, grid.to, grid.b, grid.p, station, bus_a)
    ref = grid.split(station, bus_a)
    h_ref = ref.ptdf()
    nb = grid.n_bus
    return {
        'ptdf': float(np.abs(s.h - h_ref).max()),
        'flows': float(np.abs(s.f - h_ref @ p_ext).max()),
        'beta_vs_pedf': float(np.abs(s.beta - (h_ref[:, station] - h_ref[:, nb])).max()),
        'f_plus_c_beta_vs_h_p': float(np.abs(grid.flows() + s.c * s.beta - s.h @ p_ext).max()),
    }


def reassign(grid: DcGrid, station: int, bus_a: list[int], mw: float) -> dict[str, float]:
    """Move ``mw`` of injection from bus A to bus B after the split and compare delta F with -beta x (Lab step 2)."""
    s, p_ext = split(grid.ptdf(), grid.frm, grid.to, grid.b, grid.p, station, bus_a)
    nb = grid.n_bus
    p_moved = p_ext.copy()
    p_moved[station] -= mw
    p_moved[nb] += mw
    delta = s.h @ p_moved - s.f
    ref = grid.split(station, bus_a).moved(station, nb, mw).flows()
    return {
        'delta_f_vs_minus_beta_x': float(np.abs(delta + s.beta * mw).max()),
        'vs_recomputed': float(np.abs(s.h @ p_moved - ref).max()),
        'largest_change_mw': float(np.abs(delta).max()),
    }


def beta_sum(grid: DcGrid, station: int, bus_a: list[int]) -> float:
    """Exercise 3: sum of the bus-A branches' beta entries oriented out of bus A (expected: exactly one)."""
    s, _ = split(grid.ptdf(), grid.frm, grid.to, grid.b, grid.p, station, bus_a)
    sign = np.array([1.0 if grid.frm[k] == station else -1.0 for k in bus_a])
    return float(sign @ s.beta[bus_a])


def admissible_splits(branches: list[int], minimum: int = 2) -> list[list[int]]:
    """Two-way splits with at least ``minimum`` branches per busbar; a split and its mirror image count once."""
    first, rest = branches[0], branches[1:]
    out = []
    for r in range(minimum - 1, len(branches) - minimum):  # bus A holds r + 1 branches, bus B the rest
        out += [[first, *c] for c in combinations(rest, r)]  # the first branch is always on bus A
    return out


def scan_station(grid: DcGrid, station: int, limits: np.ndarray) -> list[tuple[list[str], float, float]]:
    """Exercise 4: (bus-A branches, |c| in MW, worst post-split loading in %) for every admissible split."""
    h = grid.ptdf()
    rows = []
    for bus_a in admissible_splits(grid.branches_at(station)):
        s, _ = split(h, grid.frm, grid.to, grid.b, grid.p, station, bus_a)
        rows.append(([grid.names[k] for k in bus_a], abs(s.c), float((100 * np.abs(s.f) / limits).max())))
    return rows


def main() -> None:

    grid = load_case14()
    station = 1  # bus 2: generator 40 MW and load 21.7 MW; branches L1-2, L2-3, L2-4, L2-5
    bus_a = [grid.names.index('L1-2'), grid.names.index('L2-5')]
    print('bus 2 split, bus A = L1-2 + L2-5 (with the injections), bus B = L2-3 + L2-4')
    for key, value in check_split(grid, station, bus_a).items():
        print(f'  max |difference| {key:22s} {value:.2e}')
    s, _ = split(grid.ptdf(), grid.frm, grid.to, grid.b, grid.p, station, bus_a)
    print(f'  cross-coupler flow c = {s.c:.2f} MW from bus A to bus B')
    frm2, to2 = moved_ends(grid.frm, grid.to, station, bus_a, grid.n_bus)
    for k, name in enumerate(grid.names):
        if station in (frm2[k], to2[k]) or grid.n_bus in (frm2[k], to2[k]):
            print(f'  {name:6s} flow {grid.flows()[k]:8.2f} -> {s.f[k]:8.2f} MW, beta {s.beta[k]:+.4f}')
    for key, value in reassign(grid, station, bus_a, 40.0).items():

    print('Exercise 4: every admissible split of bus 4 (L2-4, L3-4, L4-5, T4-7, T4-9), limits 1.3 x base, 40 MW floor')
    limits = ratings(grid)
    print(f'  unsplit worst loading {(100 * np.abs(grid.flows()) / limits).max():.1f} %')
    for names, c, loading in sorted(scan_station(grid, 3, limits), key=lambda r: r[1]):
        print(f'  bus A = {"+".join(names):22s} |c| = {c:6.1f} MW  worst loading {loading:6.1f} %')

      modf.py
      """Multiple-outage distribution factors and the Woodbury susceptance update of Chapter 18, in NumPy.

Notation of the chapter: ``h`` is the PTDF (branches x nodes), ``frm``/``to`` the branch ends, and for a set of
branches O the matrix script-H[:, O] = h[:, frm_O] - h[:, to_O] holds the flow response to one MW sent across each
branch of O; its k x k block on the rows of O is script-H[O, O]. ``ieee14.load_case14()`` gives a grid to try them on.
"""

import numpy as np

ISLAND_TOL = 1e-10  # ToOp: |det| above this means the outage set leaves the grid connected


def transfer(h: np.ndarray, frm: np.ndarray, to: np.ndarray, branches: list[int]) -> np.ndarray:
    """script-H[:, O]: PTDF columns of the from-ends minus those of the to-ends (branches x k)."""
    return h[:, frm[branches]] - h[:, to[branches]]


def modf(h: np.ndarray, frm: np.ndarray, to: np.ndarray, outages: list[int]) -> tuple[np.ndarray, float]:
    """MODF (branches x k) of a joint outage of ``outages`` and det(I - script-H[O, O]).

    A determinant below ``ISLAND_TOL`` means the outage set islands the grid; the MODF is then all NaN.
    """
    n_t = transfer(h, frm, to, outages)
    m = np.eye(len(outages)) - n_t[outages]
    det = float(np.linalg.det(m))
    if abs(det) < ISLAND_TOL:
        return np.full_like(n_t, np.nan), det
    n_t = n_t.copy()
    n_t[outages] = -m  # the outaged branches lose their own flow
    return np.linalg.solve(m.T, n_t.T).T, det  # N (I - script-H[O, O])^{-1}


def apply_modf(h: np.ndarray, frm: np.ndarray, to: np.ndarray, outages: list[int]) -> np.ndarray:
    """PTDF after the joint outage: H + MODF H[O, :], rows O zero (equation 7 of Guo et al.)."""
    m, det = modf(h, frm, to, outages)
    if abs(det) < ISLAND_TOL:
        raise ValueError(f'the outage of {outages} islands the grid (det = {det:.2e})')
    h_new = h + m @ h[outages]
    h_new[outages] = 0.0
    return h_new


def modf_flows(f: np.ndarray, h: np.ndarray, frm: np.ndarray, to: np.ndarray, outages: list[int]) -> np.ndarray:
    """Flows after the joint outage: F + MODF F[O], with F[O] = 0 afterwards."""
    m, _ = modf(h, frm, to, outages)
    f_new = f + m @ f[outages]
    f_new[outages] = 0.0
    return f_new


def woodbury(h: np.ndarray, frm: np.ndarray, to: np.ndarray, branches: list[int], alpha: np.ndarray) -> np.ndarray:
    """PTDF after changing the susceptances of ``branches`` by the relative amounts ``alpha`` (delta b / b).

    H' = H + (E_O - script-H[:, O]) C with C = (I + diag(alpha) script-H[O, O])^{-1} diag(alpha) H[O, :]. With every
    alpha equal to -1 (susceptance to zero) it is the MODF update.
    """
    alpha = np.asarray(alpha, float)
    n_t = transfer(h, frm, to, branches)
    m = np.eye(len(branches)) + alpha[:, None] * n_t[branches]
    c = np.linalg.solve(m, alpha[:, None] * h[branches])
    e = np.zeros_like(n_t)
    e[branches, np.arange(len(branches))] = 1.0
    return h + (e - n_t) @ c


def lodf_matrix(h: np.ndarray, frm: np.ndarray, to: np.ndarray) -> np.ndarray:
    """LODF matrix (branches x outaged branches); columns of bridges are NaN; diagonal -1."""
    n_t = transfer(h, frm, to, list(range(h.shape[0])))
    den = 1.0 - np.diag(n_t)
    with np.errstate(divide='ignore', invalid='ignore'):
        lodf = np.where(np.abs(den) < 1e-11, np.nan, n_t / den)
    np.fill_diagonal(lodf, -1.0)
    return lodf


def modf_2x2(h: np.ndarray, frm: np.ndarray, to: np.ndarray, k1: int, k2: int) -> tuple[np.ndarray, float]:
    """Exercise 1: the MODF of two branches with the closed-form inverse of (I - script-H[O, O]).

    With a = script-H[k1, k1], b = script-H[k1, k2], c = script-H[k2, k1], d = script-H[k2, k2]:
    I - script-H[O, O] = [[1 - a, -b], [-c, 1 - d]] has determinant (1 - a)(1 - d) - b c and inverse
    [[1 - d, b], [c, 1 - a]] / det. Returns the MODF and the determinant.
    """
    n_t = transfer(h, frm, to, [k1, k2])
    a, b = n_t[k1]
    c, d = n_t[k2]
    det = (1 - a) * (1 - d) - b * c
    inv = np.array([[1 - d, b], [c, 1 - a]]) / det
    n_t = n_t.copy()
    n_t[[k1, k2]] = -np.array([[1 - a, -b], [-c, 1 - d]])
    return n_t @ inv, float(det)

    )
    t = grid.names.index('T5-6')

      """Compose ToOp's low-rank updates on a DC grid: bus split, disconnection, N-1 screen and busbar outage (Chapter 18).

The functions take the arrays of an ``ieee14.DcGrid`` (or of ToOp's ``NetworkData``) and follow the order of
``compute_symmetric_batch``: bus splits, then disconnections, then the contingency factors on the fully updated
PTDF; on the flow side the cross-coupler flows, then the disconnection update, then the contingency flows.
"""

from dataclasses import dataclass, replace

import numpy as np
from bsdf import bsdf_split
from modf import ISLAND_TOL, apply_modf, lodf_matrix, modf, modf_flows


def coupler_row(h: np.ndarray, frm: np.ndarray, to: np.ndarray, i: int, bus_a: list[int]) -> np.ndarray:
    """The row p_c with c = p_c P: KCL at bus A, counting to-end branches positive (``i`` is not the slack)."""
    a_from = [k for k in bus_a if frm[k] == i]
    a_to = [k for k in bus_a if to[k] == i]
    p_c = h[a_to].sum(0) - h[a_from].sum(0)
    p_c[i] += 1.0
    return p_c


@dataclass(frozen=True)
class Split:
    """The outcome of one bus split: updated PTDF, flows, beta, the coupler row and the cross-coupler flow."""

    h: np.ndarray
    f: np.ndarray
    beta: np.ndarray
    p_c: np.ndarray
    c: float


def split(h: np.ndarray, frm: np.ndarray, to: np.ndarray, b: np.ndarray, p: np.ndarray, i: int, bus_a: list[int]):
    """Split node ``i``; the new node is appended as the last column of the PTDF and the last injection (zero).

    ``h`` is a PTDF with as many columns as ``p`` has entries. Returns a ``Split`` plus the extended injections.
    """
    nb = h.shape[1]
    h_ext = np.hstack([h, h[:, [i]]])
    p_ext = np.append(p, 0.0)
    p_c = coupler_row(h_ext, frm, to, i, bus_a)
    h_new, f_new = bsdf_split(h_ext, frm, to, b, i, nb, bus_a, p_ext)
    return Split(h=h_new, f=f_new, beta=h_new[:, i] - h_new[:, nb], p_c=p_c, c=float(p_c @ p_ext)), p_ext


def moved_ends(frm: np.ndarray, to: np.ndarray, i: int, bus_a: list[int], nb: int) -> tuple[np.ndarray, np.ndarray]:
    """From/to vectors after the split: the branches of ``i`` that are not in ``bus_a`` now end at ``nb``."""
    frm, to = frm.copy(), to.copy()
    for k in range(len(frm)):
        if k not in bus_a:
            if frm[k] == i:
                frm[k] = nb
            elif to[k] == i:
                to[k] = nb
    return frm, to


def n1_screen(h: np.ndarray, f: np.ndarray, frm: np.ndarray, to: np.ndarray, outages: list[int]) -> np.ndarray:
    """Post-contingency flows (outages x branches) from the LODF matrix; a bridge outage gives a NaN row."""
    lodf = lodf_matrix(h, frm, to)
    return f[None, :] + lodf[:, outages].T * f[outages][:, None]


def busbar_outage(h, f, frm, to, p, node) -> tuple[np.ndarray, list[int]]:
    """Flows after losing a busbar: its injection goes first (PTDF column), then all its branches (MODF).

    If the MODF of all branches islands the grid, retry without the first branch, which stays as a skeleton
    branch (ToOp's ``perform_outage_single_busbar``). Returns the flows and the branches actually outaged; NaN
    flows if both attempts fail.
    """
    at_node = [k for k in range(len(frm)) if frm[k] == node or to[k] == node]
    f_hat = f - h[:, node] * p[node]
    for branches in (at_node, at_node[1:]):
        if not branches:
            return f_hat, []
        if abs(modf(h, frm, to, branches)[1]) >= ISLAND_TOL:
            return modf_flows(f_hat, h, frm, to, branches), branches
    return np.full_like(f, np.nan), at_node


@dataclass(frozen=True)
class Composed:
    """Everything the composed computation produces, for comparison with the brute-force references."""

    h: np.ndarray
    f: np.ndarray
    n1: np.ndarray
    f_bb: np.ndarray
    beta: np.ndarray
    c: float


def compose(grid, station: int, bus_a: list[int], disconnect: int, busbar: int, split_first: bool = True) -> Composed:
    """Bus split, then a disconnection, then an N-1 screen with a busbar outage, in ToOp's order (or swapped).

    ``grid`` is an ``ieee14.DcGrid``. With ``split_first=False`` the disconnection is applied before the split,
    the alternative order of Lab step 5. The N-1 screen outages every branch except the disconnected one.
    """
    h, p, frm, to = grid.ptdf(), grid.p, grid.frm, grid.to
    nb = grid.n_bus
    if split_first:
        s, p = split(h, frm, to, grid.b, p, station, bus_a)
        frm, to = moved_ends(frm, to, station, bus_a, nb)
        h = apply_modf(s.h, frm, to, [disconnect])
        f = modf_flows(s.f, s.h, frm, to, [disconnect])
    else:
        h_d = apply_modf(h, frm, to, [disconnect])
        s, p = split(h_d, frm, to, grid.b, p, station, bus_a)
        frm, to = moved_ends(frm, to, station, bus_a, nb)
        h, f = s.h, s.f
        f[disconnect] = 0.0
    frm, to = frm.copy(), to.copy()
    frm[disconnect] = to[disconnect] = -1  # ToOp marks a disconnected branch with an invalid node index
    outages = [k for k in range(grid.n_branch) if k != disconnect]
    n1 = n1_screen(h, f, frm, to, outages)
    f_bb, _ = busbar_outage(h, f, frm, to, p, busbar)
    return Composed(h=h, f=f, n1=n1, f_bb=f_bb, beta=s.beta, c=s.c)


def brute_force(grid, station: int, bus_a: list[int], disconnect: int, busbar: int):
    """The same N-0, N-1 and busbar-outage flows from PTDFs recomputed on the physically modified grids."""
    base = grid.split(station, bus_a).without([disconnect])
    f = base.flows()
    n1 = [np.insert(base.without([k]).flows(), k, 0.0) for k in range(base.n_branch)]
    at_busbar = base.branches_at(busbar)
    p = base.p.copy()
    p[busbar] = 0.0
    f_bb = np.zeros(base.n_branch)
    keep = [k for k in range(base.n_branch) if k not in at_busbar]
    f_bb[keep] = replace(base.without(at_busbar), p=p).flows()
    return f, np.array(n1), f_bb

    Exercise 1¶

    Let \(O = \{k_1, k_2\}\) and write the \(2 \times 2\) block as

    \[ \mathcal H_{OO} = \begin{pmatrix} a & b \\ c & d \end{pmatrix}, \qquad I - \mathcal H_{OO} = \begin{pmatrix} 1 - a & -b \\ -c & 1 - d \end{pmatrix}, \qquad \det = (1 - a)(1 - d) - bc, \]
    \[ (I - \mathcal H_{OO})^{-1} = \frac{1}{\det} \begin{pmatrix} 1 - d & b \\ c & 1 - a \end{pmatrix}, \qquad \text{MODF} = N\,(I - \mathcal H_{OO})^{-1}, \]

    with \(N = \mathcal H_{:,O}\) except rows \(k_1, k_2\), which are \(-(I - \mathcal H_{OO})\) so that each outaged branch loses its own flow (modf_2x2). For \(b = c = 0\) the two columns reduce to the two single LODF columns \(\mathcal H_{:,k}/(1 - h_{kk})\).

    Cut set. Orient both branches from the side \(S\) to the side \(S^c\) that the cut separates. One MW injected at the from-end of \(k_j\) (in \(S\)) and withdrawn at its to-end (in \(S^c\)) must cross the cut in full, and the only branches crossing it are \(k_1\) and \(k_2\), so the column sums of \(\mathcal H_{OO}\) are one: \(\mathbf 1^\top \mathcal H_{OO} = \mathbf 1^\top\), hence \(\mathbf 1^\top (I - \mathcal H_{OO}) = 0\) and the determinant vanishes. A bridge is the case \(k = 1\) (\(h_{kk} = 1\)); for a cut set of two the diagonal entries are below one, since one MW across \(k_1\) partly returns through \(k_2\). On IEEE 14, L1-2 and L1-5 are the two branches of bus 1: \(1 - h_{kk} = 0.162\) and \(0.611\) (neither is a bridge), the column sums of \(\mathcal H_{OO}\) are \([1, 1]\) and \(\det = 7 \cdot 10^{-16}\), below ToOp's \(10^{-10}\). For comparison, L2-4 with L4-5 (not a cut) gives \(\det = 0.0802\) (modf.py, main()).

    """Exercise 2 of Chapter 18: the Woodbury susceptance update on IEEE 118 against recomputing the PTDF.

``check(grid, k)`` verifies the update for k simultaneous susceptance changes; ``timing(grid, ks)`` times both
routes. Timings depend on the machine and on the BLAS build, the crossover point less so.

import timeit

import numpy as np
import pandapower.networks as pn
from ieee14 import DcGrid, from_pandapower, ptdf
from modf import woodbury


def changed(grid: DcGrid, branches: list[int], alpha: list[float]) -> DcGrid:
    """The grid with the susceptances of ``branches`` multiplied by ``1 + alpha``."""
    b = grid.b.copy()
    b[branches] *= 1 + np.asarray(alpha)
    return DcGrid(frm=grid.frm, to=grid.to, b=b, p=grid.p, names=grid.names, slack=grid.slack)


def check(grid: DcGrid, k: int, seed: int = 0) -> float:
    """Largest difference between the Woodbury update and the recomputed PTDF for ``k`` random changes."""
    rng = np.random.default_rng(seed)
    branches = sorted(rng.choice(grid.n_branch, size=k, replace=False).tolist())
    alpha = rng.uniform(-0.5, 0.5, size=k).tolist()
    h_w = woodbury(grid.ptdf(), grid.frm, grid.to, branches, alpha)
    return float(np.abs(h_w - changed(grid, branches, alpha).ptdf()).max())


def timing(grid: DcGrid, ks: tuple[int, ...], number: int = 20) -> list[tuple[int, float, float]]:
    """(k, Woodbury milliseconds, recompute milliseconds) per k, best of five repeats of ``number`` runs."""
    h = grid.ptdf()
    rows = []
    for k in ks:
        branches = list(range(k))
        alpha = [0.2] * k
        update = partial(woodbury, h, grid.frm, grid.to, branches, alpha)
        g = changed(grid, branches, alpha)
        recompute = partial(ptdf, g.frm, g.to, g.b, g.n_bus, g.slack)
        t_w = min(timeit.repeat(update, number=number, repeat=5))
        t_r = min(timeit.repeat(recompute, number=number, repeat=5))
        rows.append((k, 1e3 * t_w / number, 1e3 * t_r / number))
    return rows

    Exercise 3¶

    In the split network (PTDF \(H'\), injections \(P'\) with the station's injection on bus A) the flows are \(F' = H'P'\). Closing a coupler between A and B that carries \(c\) MW from A to B is, seen from the split network, an extra withdrawal of \(c\) at A and an extra injection of \(c\) at B, so the unsplit flows are

    \[ F = H'\left(P' - c\,e_i + c\,e_b\right) = F' - c\left(H'_{:,i} - H'_{:,b}\right). \]

    Rearranged, \(F' = F + c\,\beta\) with \(\beta = H'_{:,i} - H'_{:,b}\): the post-split PEDF from A to B, i.e. the flow pattern of one MW injected at A and withdrawn at B in the split network.

    That reading gives the sum rule at once. Inject one MW at bus A and take it out at bus B. KCL at bus A: the one MW leaves A through the bus-A branches, so, counting a branch positive when its from-end is at A (flow out of A) and negative when its to-end is at A,

    \[ \sum_{\ell \in A_\text{from}} \beta_\ell - \sum_{\ell \in A_\text{to}} \beta_\ell = 1 . \]

    On IEEE 14, bus 2 split as in the Lab: \(\beta_{\text{L2-5}} - \beta_{\text{L1-2}} = 0.6188 + 0.3812 = 1.000000000000\) (splits.beta_sum); test_splits.py checks every admissible split of buses 2, 4, 5 and 6. The same KCL at bus B gives minus one for the bus-B branches oriented out of B, which is why \(p_c\) and \(\beta\) together conserve power in \(H' = H + \beta p_c\).

    Exercise 5¶

    With \(n_l\) branches, \(n_b\) nodes, a split station of \(d\) branches of which \(a\) stay on bus A, \(k\) outages, \(n_f\) N-1 cases and \(T\) timesteps, counting a multiply-add as two flops and a \(k \times k\) LU solve as \(\tfrac{2}{3}k^3\) (flops.py):

    Update Flops Leading term
    one BSDF \(d\,n_b + 2a\,n_l + 4a + n_l + 2 n_l n_b\) \(2 n_l n_b\) (the rank-1 update \(H + \beta p_c\))
    one \(k\)-outage MODF \(n_l k + \tfrac{2}{3}k^3 + 2 n_l k^2 + 2 n_l n_b k\) \(2 n_l n_b k\) (the rank-\(k\) update)
    LODF matrix, \(n_f\) cases \(2 n_l n_f + n_f\) \(2 n_l n_f\) (one difference and one division per entry)
    N-1 flow matrix, \(T\) timesteps \(2 n_l n_b T + 2 n_l n_f T\) \(2 n_l (n_b + n_f) T\)
    full PTDF recomputation \(\tfrac{2}{3} n_b^3 + 2 n_l n_b^2\)

    IEEE 14 (\(n_l = 20\), \(n_b = 14\), \(d = 5\), \(a = 2\), \(k = 3\), \(n_f = 20\), \(T = 1\)): 738, 2 118, 820 and 1 360 flops against 9 669 for a recomputation. A 1 000-node grid with 1 500 branches, all of them N-1 cases, one day of hourly timesteps: \(3.0 \cdot 10^6\), \(9.0 \cdot 10^6\), \(4.5 \cdot 10^6\) and \(1.8 \cdot 10^8\) flops against \(3.7 \cdot 10^9\) for a recomputation. Every update is a few dense matrix products of fixed shape, and this is the point: the N-1 flow matrix, the largest item, is \(n_f\) independent rank-1 updates of \(T\) independent vectors, the LODF matrix is \(n_l n_f\) independent entries, and a MODF is one small solve followed by a product. Across a batch of topologies nothing is shared: each topology has its own \(\beta\), MODF, LODF matrix and flows, all with the same shapes, so the whole chain is embarrassingly parallel over the batch (a vmap in ToOp). The only sequential dependence is inside one topology: the second split of a station list needs the PTDF left by the first (compute_bus_splits), and the LODF matrix needs the PTDF after all splits and disconnections. That dependence is short (a handful of splits) and the parallel dimension is long (thousands of topologies, \(n_f\) cases, \(T\) timesteps), which is what makes the GPU the right machine for it (Chapter 32).

    flops.py
    """Exercise 5 of Chapter 18: floating-point operation counts of ToOp's low-rank updates.

A multiply-add counts as two flops, a k x k LU solve as 2k³/3 (its leading term), and gathers and sign flips as
nothing. ``n_l`` branches, ``n_b`` nodes, ``d`` branches at the split station of which ``a`` stay on bus A, ``k``
simultaneous outages, ``n_f`` N-1 cases and ``t`` timesteps.
"""


def bsdf_flops(n_l: int, n_b: int, d: int, a: int) -> int:
    """One BSDF update: coupler row, numerator, denominator, beta and the rank-1 PTDF update."""
    coupler_row = d * n_b  # sum of d PTDF rows
    numerator = 2 * a * n_l  # a scaled differences of PTDF columns
    denominator = 4 * a  # two sums over bus-A branches
    beta = n_l  # one division per branch
    update = 2 * n_l * n_b  # H + beta p_c
    return coupler_row + numerator + denominator + beta + update


def modf_flops(n_l: int, n_b: int, k: int) -> int:
    """One k-outage MODF: the k x k system, the n_l x k MODF matrix and the rank-k PTDF update."""
    transfer = n_l * k  # script-H[:, O] = H[:, f] - H[:, t]
    solve = 2 * k**3 // 3  # LU of I - script-H[O, O]
    modf_matrix = 2 * n_l * k * k  # N (I - script-H[O, O])^{-1}
    update = 2 * n_l * n_b * k  # H + MODF H[O, :]
    return transfer + solve + modf_matrix + update


def lodf_matrix_flops(n_l: int, n_f: int) -> int:
    """The LODF matrix for n_f single outages: one difference and one division per entry, n_f denominators."""
    return 2 * n_l * n_f + n_f


def n1_flows_flops(n_l: int, n_b: int, n_f: int, t: int) -> int:
    """N-0 flows H P for t timesteps and the N-1 matrix F + LODF[:, k] F_k for n_f outages each."""
    return 2 * n_l * n_b * t + 2 * n_l * n_f * t


def table(
    n_l: int, n_b: int, d: int = 5, a: int = 2, k: int = 3, n_f: int | None = None, t: int = 1
) -> dict[str, int]:
    """Every count for one grid size; ``n_f`` defaults to all branches."""
    n_f = n_l if n_f is None else n_f
    return {
        'BSDF update (one split)': bsdf_flops(n_l, n_b, d, a),
        f'MODF update (k = {k})': modf_flops(n_l, n_b, k),
        f'LODF matrix (n_f = {n_f})': lodf_matrix_flops(n_l, n_f),
        f'N-1 flow matrix (T = {t})': n1_flows_flops(n_l, n_b, n_f, t),
        'full PTDF recomputation': 2 * n_b**3 // 3 + 2 * n_l * n_b * n_b,
    }


def main() -> None:
    """Counts for IEEE 14 and for a transmission grid of ToOp's size."""
    for label, n_l, n_b, t in (('IEEE 14', 20, 14, 1), ('a 1 000-node grid, one day', 1500, 1000, 24)):
        print(label)
        for name, flops in table(n_l, n_b, t=t).items():
            print(f'  {name:28s} {flops:14,d} flops')


if __name__ == '__main__':
    main()

    Exercise 6¶

    bridges.random_grid builds a random spanning tree on 12 nodes, adds 8 random branches, and hangs a leaf node on one radial branch \(k\) with random susceptances in \([2, 20]\); one_minus_hkk computes \(1 - h_{kk}\) from a PTDF obtained by an LU solve. Over 2 000 grids (survey(2000)), \(1 - h_{kk}\) is exactly zero in 74.2 % of them; the median of \(|1 - h_{kk}|\) is 0 and the maximum \(3.3 \cdot 10^{-16}\), a few units in the last place. In exact arithmetic \(h_{kk} = 1\) for a bridge, because the one MW sent across it has no other way; in floating point the solve returns the row of \(H\) with round-off, and whether the subtraction cancels to zero depends on the bit pattern, not on the physics.

    Which of ToOp's tolerances catch the bridge:

    Check Catches the bridge in
    LODF, grid connected only if \(\lvert 1 - h_{kk} \rvert > 10^{-11}\) 100 % of the grids
    BSDF, \(\lvert d \rvert \ge 10^{-5}\) 100 %
    \(2 \times 2\), \(3 \times 3\) and LU solves, \(\lvert \det \rvert > 10^{-10}\) 100 %
    \(1 \times 1\) solve_and_check_det, a != 0 exactly 74.2 %

    Any relative or absolute tolerance far above machine precision (\(10^{-11}\), \(10^{-10}\), \(10^{-5}\)) catches every bridge here, since the round-off is at the \(10^{-16}\) level. The exact test of the \(1 \times 1\) case lets one bridge in four through: the division by \(a \approx 10^{-16}\) then produces flows of order \(10^{16}\) MW instead of a failure flag. This is the "inconsistent islanding tolerances" discrepancy of the chapter, and the lesson of Chapter 4: never test a floating-point result for exact zero; compare it with a tolerance that is large against the round-off of the computation that produced it and small against the values that mean something.

    bridges.py
    """Exercise 6 of Chapter 18: how often is 1 - h_kk exactly zero for a radial branch of a random meshed grid?

Each grid is a random spanning tree on ``n`` nodes with extra random branches, plus one leaf node hung on a single
radial branch k. Mathematically h_kk = 1 for a bridge, so 1 - h_kk = 0; in floating point the PTDF comes out of
an LU solve and the difference is round-off instead. The chapter lists ToOp's islanding tolerances; the exercise
asks which of them would notice this bridge.
"""

import numpy as np
from ieee14 import ptdf

TOLERANCES = {
    'LODF: |1 - h_kk| > 1e-11 means connected': 1e-11,
    'BSDF: |d| >= 1e-5': 1e-5,
    '2x2, 3x3 and LU solves: |det| > 1e-10': 1e-10,
    '1x1 solve_and_check_det: a != 0 (exact)': 0.0,
}


def random_grid(n: int, extra: int, rng: np.random.Generator) -> tuple[np.ndarray, np.ndarray, np.ndarray, int]:
    """From/to/susceptance arrays of a meshed grid on ``n`` nodes plus a leaf; returns the radial branch index."""
    frm, to = [], []
    for node in range(1, n):  # random spanning tree: each new node attaches to an earlier one
        frm.append(int(rng.integers(0, node)))
        to.append(node)
    while len(frm) < n - 1 + extra:  # extra meshing branches between distinct nodes
        i, j = rng.integers(0, n, size=2)
        if i != j:
            frm.append(int(i))
            to.append(int(j))
    frm.append(int(rng.integers(0, n)))  # the radial branch to the leaf node n
    to.append(n)
    b = rng.uniform(2.0, 20.0, size=len(frm))
    return np.array(frm), np.array(to), b, len(frm) - 1


def one_minus_hkk(frm: np.ndarray, to: np.ndarray, b: np.ndarray, k: int) -> float:
    """1 - h_kk of branch ``k``, with h_kk the flow on k when one MW is sent from its from-end to its to-end."""
    h = ptdf(frm, to, b, int(max(frm.max(), to.max())) + 1)
    return float(1.0 - (h[k, frm[k]] - h[k, to[k]]))


def survey(trials: int, n: int = 12, extra: int = 8, seed: int = 0) -> np.ndarray:
    """1 - h_kk of the radial branch for ``trials`` random grids."""
    rng = np.random.default_rng(seed)
    return np.array([one_minus_hkk(*random_grid(n, extra, rng)) for _ in range(trials)])


def main() -> None:
    """Survey 2 000 grids and say which tolerance catches the bridge in every one of them."""
    values = survey(2000)
    exact = np.mean(values == 0.0)
    print(f'{len(values)} random grids, 12 meshed nodes + 1 leaf: 1 - h_kk is exactly zero in {100 * exact:.1f} %')
    print(f'  |1 - h_kk| median {np.median(np.abs(values)):.1e}, max {np.abs(values).max():.1e}')
    for name, tol in TOLERANCES.items():
        caught = np.mean(np.abs(values) <= tol) if tol > 0 else exact
        print(f'  {name:45s} catches the bridge in {100 * caught:5.1f} % of the grids')


if __name__ == '__main__':
    main()
