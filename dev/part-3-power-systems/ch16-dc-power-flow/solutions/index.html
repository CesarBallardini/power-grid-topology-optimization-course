    1. ToOp's example notebook. toop57.py does what example1_dc_loadflow_example.ipynb does: case57_data_powsybl(folder) writes the grid and masks, load_grid(..., pandapower=False) preprocesses them and run_initial_loadflow fills unsplit_flow.       toop57.py
      """ToOp's base-case DC flows (``unsplit_flow``) on its IEEE 57 example against pypowsybl (Chapter 16, Lab steps 1, 4).

Run it with ToOp's interpreter (its container or virtual environment)::

    python toop57.py /tmp/case57

It writes ToOp's extended IEEE 57 case in PowSyBl format into that folder, as
``notebooks/example1_dc_loadflow_example.ipynb`` does, loads it, runs the initial load flow and compares
``unsplit_flow`` with ``pypowsybl.loadflow.run_dc`` on the saved grid.
"""

import sys
from pathlib import Path

import jax
import numpy as np
import pandas as pd
import pypowsybl as pyp
from fsspec.implementations.dirfs import DirFileSystem
from toop_engine_dc_solver.example_grids import CGMES_DISTRIBUTED_SLACK, case57_data_powsybl
from toop_engine_dc_solver.preprocess.convert_to_jax import load_grid, run_initial_loadflow
from toop_engine_interfaces.folder_structure import PREPROCESSING_PATHS
from toop_engine_interfaces.messages.preprocess.preprocess_commands import PreprocessParameters

jax.config.update('jax_enable_x64', True)


def load(folder: Path):
    """Write the example case into ``folder`` and preprocess it; return static information and network data."""
    case57_data_powsybl(folder)
    _, si, network_data = load_grid(
        DirFileSystem(str(folder)),
        pandapower=False,
        parameters=PreprocessParameters(action_set_filter_bsdf_lodf=False),
    )
    si, _ = run_initial_loadflow(si)
    return si, network_data


def toop_flows(si, network_data) -> pd.DataFrame:
    """ToOp's ``unsplit_flow`` of timestep 0 next to the backend's DC flows and AC-DC mismatch, by branch id (MW)."""
    ids = list(network_data.branch_ids)
    return pd.DataFrame(
        {
            'name': list(network_data.branch_names),
            'unsplit_flow': np.asarray(si.dynamic_information.unsplit_flow)[0],
            'basecase_dc': np.asarray(network_data.basecase_dc_branch_flows)[0],
            'ac_dc_mismatch': np.asarray(network_data.ac_dc_mismatch)[0],
        },
        index=ids,
    )


def powsybl_flows(folder: Path) -> pd.Series:
    """``p1`` of every line and transformer after ``run_dc`` with ToOp's parameters on the saved grid (MW)."""
    net = pyp.network.load(str(folder / PREPROCESSING_PATHS['grid_file_path_powsybl']))
    pyp.loadflow.run_dc(net, CGMES_DISTRIBUTED_SLACK)
    return pd.concat([net.get_lines()['p1'], net.get_2_windings_transformers()['p1']]).rename('powsybl_p1')


def compare(folder: Path) -> tuple[pd.DataFrame, object]:
    """ToOp flows next to pypowsybl's ``p1``; ``unsplit + p1`` is zero when the orientations are opposite."""
    si, network_data = load(folder)
    frame = toop_flows(si, network_data)
    frame['powsybl_p1'] = powsybl_flows(folder).reindex(frame.index)
    frame['unsplit + p1'] = frame['unsplit_flow'] + frame['powsybl_p1']
    frame['unsplit - basecase_dc'] = frame['unsplit_flow'] - frame['basecase_dc']
    return frame, si


def main() -> None:
    """Shape of ``unsplit_flow`` and the agreement with pypowsybl's DC flows."""
    folder = Path(sys.argv[1] if len(sys.argv) > 1 else 'case57')
    pd.set_option('display.width', 140)
    frame, si = compare(folder)
    di = si.dynamic_information
    shape = np.asarray(di.unsplit_flow).shape
    print(f'unsplit_flow shape {shape} = (n_timesteps, n_branches): {di.n_timesteps} timestep(s), ', end='')
    print(f'{di.n_branches} branches, {di.n_nodes} nodes')
    print(f'nodal_injections shape {np.asarray(di.nodal_injections).shape}; ', end='')
    print(f'{frame.index.nunique()} distinct branch ids, {frame["name"].nunique()} distinct names')
    print(f'max |unsplit_flow - basecase_dc_branch_flows| = {frame["unsplit - basecase_dc"].abs().max():.2e} MW')
    print(f'max |unsplit_flow + p1| = {frame["unsplit + p1"].abs().max():.4f} MW (p1 has the opposite orientation)')
    mism = frame['ac_dc_mismatch'].abs()
    print(
        f'AC-DC mismatch stored for alpha: mean |m| {mism.mean():.2f} MW, max {mism.max():.2f} MW on {mism.idxmax()}'
    )
    print(frame.loc[frame['unsplit_flow'].abs().nlargest(5).index].round(3))


if __name__ == '__main__':
    main()

DC flows against rundcpp on case57. dc_vs_ac.dc_case reads pandapower's internal MATPOWER-style branch table (net._ppc['branch']: from, to, BR_X, TAP, SHIFT and the flow PF in MW) after rundcpp, and the injections as -net.res_bus.p_mw, which are balanced because the DC model has no losses. The case has 80 branches, 15 of them transformers with a tap ratio, no phase shift, slack at bus 0. With \(b = 1/x\) the worked example's function differs from rundcpp by up to 1.286 MW, all of it on transformers (the largest on trafo 12, tap 0.895: \(-30.29\) against \(-31.57\) MW). With \(b = 1/(x\tau)\) the difference is \(2.2 \times 10^{-13}\) MW. That is the systematic difference to find: MATPOWER's and pandapower's DC model divide the series susceptance by the tap ratio (manual eq. 3.23), and a tap of 0.895 changes the branch susceptance by 12 %. The per-unit base is pandapower's sn_mva (100 MVA), already applied inside _ppc; the orientation is pandapower's own, from bus to to bus for lines and hv to lv for transformers, which is why res_line.p_from_mw and res_trafo.p_hv_mw are the columns to compare.       dc_vs_ac.py
      """DC flows of a pandapower case against ``rundcpp`` and ``runpp`` (Chapter 16, Lab steps 2 and 3).

The branch table (from, to, x, tap, shift) comes from the internal MATPOWER-style structure that pandapower builds
for its own solvers (``net._ppc``), so the per-unit base is pandapower's ``sn_mva`` and the orientation is
pandapower's (lines: from bus to to bus; transformers: hv bus to lv bus).
"""

import numpy as np
import pandapower as pp
import pandapower.networks as pn
import pandas as pd
from dc_power_flow import dc_power_flow

F_BUS, T_BUS, BR_R, BR_X, TAP, SHIFT, PF = 0, 1, 2, 3, 8, 9, 13


def branch_table(net: pp.pandapowerNet) -> pd.DataFrame:
    """Branches of a solved network: from, to, r, x, tap, shift (deg) and the DC flow pandapower computed (MW)."""
    br = net._ppc['branch'].real
    names = [f'line {i}' for i in net.line.index] + [f'trafo {i}' for i in net.trafo.index]
    table = pd.DataFrame(
        {
            'from': br[:, F_BUS].astype(int),
            'to': br[:, T_BUS].astype(int),
            'r': br[:, BR_R],
            'x': br[:, BR_X],
            'tap': np.where(br[:, TAP] == 0, 1.0, br[:, TAP]),
            'shift': br[:, SHIFT],
            'p_pandapower': br[:, PF],
        },
        index=names,
    )
    table['kind'] = ['line'] * len(net.line) + ['trafo'] * len(net.trafo)
    return table


def dc_case(name: str) -> tuple[pp.pandapowerNet, pd.DataFrame, np.ndarray, int]:
    """Run ``rundcpp`` on ``pandapower.networks.<name>()``: the net, the branch table, injections (MW) and slack."""
    net = getattr(pn, name)()
    pp.rundcpp(net)
    table = branch_table(net)
    p = -net.res_bus.p_mw.to_numpy()  # res_bus is a demand: minus it is the net injection, balanced in DC
    return net, table, p, int(net.ext_grid.bus.iloc[0])


def own_dc_flows(table: pd.DataFrame, p: np.ndarray, slack: int, with_taps: bool = True) -> np.ndarray:
    """DC flows (MW) with the worked example's function, with or without the tap ratio in b = 1/(x tau)."""
    x = table['x'].to_numpy() * (table['tap'].to_numpy() if with_taps else 1.0)
    return dc_power_flow(table['from'].to_numpy(), table['to'].to_numpy(), x, p, slack)


def dc_ac_frame(name: str) -> pd.DataFrame:
    """One row per branch: DC flow, AC flow, AC loading (%), r/x and the relative DC error against the AC flow."""
    net, table, p, slack = dc_case(name)
    table['p_dc'] = own_dc_flows(table, p, slack)
    pp.runpp(net)
    table['p_ac'] = np.concatenate([net.res_line.p_from_mw.to_numpy(), net.res_trafo.p_hv_mw.to_numpy()])
    table['loading_ac'] = np.concatenate(
        [net.res_line.loading_percent.to_numpy(), net.res_trafo.loading_percent.to_numpy()]
    )
    table['losses_ac'] = np.concatenate([net.res_line.pl_mw.to_numpy(), net.res_trafo.pl_mw.to_numpy()])
    table['r_over_x'] = table['r'] / table['x']
    table['abs_error'] = (table['p_dc'] - table['p_ac']).abs()
    table['rel_error'] = table['abs_error'] / table['p_ac'].abs()
    return table


def main() -> None:

    Exercise 1¶

    With \(\Delta\theta = \theta_f - \theta_t\), the AC active flow of a lossless branch is \(p_f = |V_f||V_t|\sin(\Delta\theta)/x\) and the DC flow is \(p_f^{DC} = \Delta\theta/x\). Their ratio does not depend on \(x\):

    \[ \frac{p_f^{DC}}{p_f} = \frac{\Delta\theta}{|V_f||V_t|\sin\Delta\theta}, \qquad \varepsilon = \frac{p_f^{DC} - p_f}{p_f} = \frac{1}{|V_f||V_t|}\cdot\frac{\Delta\theta}{\sin\Delta\theta} - 1 . \]

    Both factors are monotone on the given ranges: \(\Delta\theta/\sin\Delta\theta\) grows from 1 at \(\Delta\theta \to 0\) to \(1/\operatorname{sinc}(10^\circ) = 1/0.99493 = 1.0051\) at \(10^\circ\), and \(1/(|V_f||V_t|)\) lies between \(1/1.05^2 = 0.9070\) and \(1/0.95^2 = 1.1080\). Hence

    \[ -9.30\,\% = \frac{1}{1.05^2} - 1 \;\le\; \varepsilon \;\le\; \frac{1.0051}{0.95^2} - 1 = +11.37\,\% . \]

    The angle contributes only \(+0.51\) % at \(10^\circ\); the voltage magnitudes contribute about \(\pm 10\) %, which is why the DC approximation is a flat-voltage approximation before it is a small-angle one, and why a voltage profile of 0.94–1.06 pu produces the errors of Lab step 3 even with \(r = 0\). Measured against the DC flow instead (\((p_f - p_f^{DC})/p_f^{DC} = |V_f||V_t|\operatorname{sinc}\Delta\theta - 1\)) the bounds are \(-10.2\) % and \(+10.25\) %. dc_approximation.error_bounds evaluates \(\varepsilon\) on the four voltage corners and 1,001 angles and returns the same two numbers.

    dc_approximation.py
    """Error of the DC branch flow and independence of the slack (Chapter 16, Exercises 1 and 2)."""

import numpy as np
from dc_power_flow import dc_power_flow


def relative_error(vf: float, vt: float, dtheta: float) -> float:
    """(p_DC - p_AC) / p_AC for a lossless branch: |Vf||Vt| sin(d) / d is the AC flow per unit of DC flow."""
    return dtheta / (vf * vt * np.sin(dtheta)) - 1


def error_bounds(v_lo: float = 0.95, v_hi: float = 1.05, max_deg: float = 10.0) -> tuple[float, float]:
    """Smallest and largest relative error over |V| in [v_lo, v_hi] at both ends and 0 < |d| <= max_deg."""
    d = np.deg2rad(np.linspace(1e-6, max_deg, 1001))
    errors = [relative_error(vf, vt, d) for vf in (v_lo, v_hi) for vt in (v_lo, v_hi)]
    return float(np.min(errors)), float(np.max(errors))


def angles_and_flows(
    frm: np.ndarray, to: np.ndarray, x: np.ndarray, p: np.ndarray, slack: int
) -> tuple[np.ndarray, np.ndarray]:
    """Bus angles (rad, slack at 0) and branch flows of a DC power flow, both for the chosen slack."""
    n, m = len(p), len(frm)
    a = np.zeros((m, n))
    a[np.arange(m), frm], a[np.arange(m), to] = 1.0, -1.0
    bbus = a.T @ np.diag(1 / x) @ a
    keep = np.arange(n) != slack
    theta = np.zeros(n)
    theta[keep] = np.linalg.solve(bbus[np.ix_(keep, keep)], p[keep])
    return theta, dc_power_flow(frm, to, x, p, slack)


def main() -> None:
    """Exercise 1 bounds and the slack comparison of Exercise 2 on the four-bus grid of the tests."""
    lo, hi = error_bounds()
    print(f'Exercise 1: relative error of the DC flow between {100 * lo:+.2f} % and {100 * hi:+.2f} %')
    sinc = np.sin(np.deg2rad(10)) / np.deg2rad(10)
    print(f'  sinc(10 deg) = {sinc:.5f}; 1/0.95^2 - 1 = {1 / 0.95**2 - 1:+.4f}; 1/1.05^2 - 1 = {1 / 1.05**2 - 1:+.4f}')
    print(f'  angle alone at 10 deg (|V| = 1): {100 * relative_error(1, 1, np.deg2rad(10)):+.2f} %')
    frm, to = np.array([0, 0, 1, 2]), np.array([1, 2, 3, 3])
    x = np.array([0.1, 0.2, 0.1, 0.2])
    p = np.array([0.8, 0.2, -0.5, -0.5])  # pu, like the reactances
    th0, f0 = angles_and_flows(frm, to, x, p, slack=0)
    th3, f3 = angles_and_flows(frm, to, x, p, slack=3)
    print('Exercise 2: flows (pu) with slack 0:', np.round(f0, 4), ' with slack 3:', np.round(f3, 4))
    print('  angles (deg) slack 0:', np.round(np.rad2deg(th0), 3), ' slack 3:', np.round(np.rad2deg(th3), 3))
    print('  difference of the angle vectors (constant):', np.round(np.rad2deg(th3 - th0), 3))


if __name__ == '__main__':
    main()

    Exercise 2¶

    \(B_\text{bus} = A^\top\operatorname{diag}(b)A\) is a weighted Laplacian of a connected graph: symmetric, positive semidefinite, with the one-dimensional null space spanned by \(\mathbf 1\) (every row sums to zero), and its range is the set of vectors orthogonal to \(\mathbf 1\), that is, the balanced injections. For a balanced \(P\) the equation \(B_\text{bus}\theta = P\) therefore has solutions, and they form the line \(\theta^\star + c\,\mathbf 1\). Choosing a slack \(s\) and deleting its row and column picks the one solution with \(\theta_s = 0\), which is \(\theta^{(s)} = \theta^\star - \theta^\star_s\,\mathbf 1\) (the reduced matrix is nonsingular, Chapter 2). Two slacks \(s\) and \(r\) give \(\theta^{(r)} - \theta^{(s)} = (\theta^\star_s - \theta^\star_r)\,\mathbf 1\): the angle vectors differ by a constant, so the angles depend on the slack. The flows do not, because \(F = \operatorname{diag}(b)A\theta\) and every row of \(A\) has one \(+1\) and one \(-1\), so \(A\,\mathbf 1 = 0\) and \(F^{(r)} - F^{(s)} = \operatorname{diag}(b)A\,c\,\mathbf 1 = 0\). The flow of a branch is a difference of two angles, and the slack choice only adds the same constant to both. If the injections do not sum to zero there is no solution of the full system; the reduced system then assigns the imbalance to the slack bus, \(P_s = -\sum_{i\ne s}P_i\), and the flows do depend on where the imbalance is absorbed (Chapter 17).

    Numerically (dc_approximation.main), on the four-bus ring with \(P = (0.8, 0.2, -0.5, -0.5)\) pu, slack 0 and slack 3 give the same flows \((0.3333, 0.4667, 0.5333, -0.0333)\) pu while the angles move from \((0, -1.910, -5.348, -4.966)^\circ\) to \((4.966, 3.056, -0.382, 0)^\circ\): a shift of \(4.966^\circ\) on every bus.

    Exercise 3¶

    dc_sparse_solve.susceptance_matrices assembles \(A\) as a sparse (20, 14) incidence matrix from the branch table of pandapower's case14 (with \(b = 1/(x\tau)\): three transformers have taps), then \(B_f = \operatorname{diag}(b)A\) and \(B_\text{bus} = A^\top B_f\), 54 nonzeros. The slack row and column are dropped, splu factorizes the (13, 13) matrix once, and flows_many solves all right-hand sides in one call: lu.solve(p[:, keep].T) takes the (13, 1000) block and \(F = (B_{f,\text{red}}\,\Theta)^\top\) has shape (1000, 20), the layout of ToOp's unsplit_flow. The snapshots are the base case plus Gaussian noise of 10 MW per bus, rebalanced at the slack; snapshot 0 is the base case itself and matches rundcpp to \(7.5 \times 10^{-14}\) MW. The block solve and 1,000 separate spsolve calls agree exactly (difference 0.0). Time (best of several runs): 0.53 ms for the factorization plus the block solve against 38–42 ms for the separate solves, a factor of 70–80. The separate solves pay a symbolic and numeric factorization per call plus Python overhead; the block solve pays one factorization and then one triangular sweep per snapshot in compiled code. This is the PTDF idea in disguise: one topology, one factorization, many injection vectors.

    dc_sparse_solve.py
    """Sparse DC power flow of IEEE 14 for many injection snapshots at once (Chapter 16, Exercise 3).

Bbus and Bf are assembled as ``scipy.sparse`` matrices from the branch table that pandapower builds for ``case14``,
factorized once with ``splu`` and reused for a ``(n_timesteps, n_bus)`` block of balanced injections.
"""

import timeit

import numpy as np
import scipy.sparse as sps
import scipy.sparse.linalg as spla
from dc_vs_ac import dc_case


def susceptance_matrices(
    frm: np.ndarray, to: np.ndarray, b: np.ndarray, n_bus: int
) -> tuple[sps.csc_matrix, sps.csr_matrix]:
    """Sparse Bbus = Aᵀ diag(b) A and Bf = diag(b) A for branch susceptances ``b``."""
    m = len(frm)
    rows = np.concatenate([np.arange(m), np.arange(m)])
    a = sps.csr_matrix(
        (np.concatenate([np.ones(m), -np.ones(m)]), (rows, np.concatenate([frm, to]))), shape=(m, n_bus)
    )
    bf = sps.diags(b) @ a
    return (a.T @ bf).tocsc(), bf.tocsr()


def flows_many(lu: spla.SuperLU, bf_reduced: sps.csr_matrix, p: np.ndarray, keep: np.ndarray) -> np.ndarray:
    """Flows ``(n_timesteps, n_branches)`` for the injections ``p`` ``(n_timesteps, n_bus)`` with one factorization."""
    theta = lu.solve(p[:, keep].T)  # (n_bus - 1, n_timesteps), every snapshot in one call
    return (bf_reduced @ theta).T


def flows_one_by_one(
    bbus_reduced: sps.csc_matrix, bf_reduced: sps.csr_matrix, p: np.ndarray, keep: np.ndarray
) -> np.ndarray:
    """The same flows with a separate ``spsolve`` (and therefore a separate factorization) per snapshot."""
    return np.array([bf_reduced @ spla.spsolve(bbus_reduced, row[keep]) for row in p])


def main(n_timesteps: int = 1000) -> None:
    """Build IEEE 14, draw balanced snapshots, time the two approaches and check that they agree."""
    _, table, p0, slack = dc_case('case14')
    n_bus = int(max(table['from'].max(), table['to'].max())) + 1
    b = 1 / (table['x'].to_numpy() * table['tap'].to_numpy())
    bbus, bf = susceptance_matrices(table['from'].to_numpy(), table['to'].to_numpy(), b, n_bus)
    keep = np.arange(n_bus) != slack
    bbus_r, bf_r = bbus[keep][:, keep], bf[:, keep]
    rng = np.random.default_rng(16)
    p = p0 + rng.normal(scale=10.0, size=(n_timesteps, n_bus))
    p[0] = p0  # snapshot 0 is the base case itself
    p[:, slack] -= p.sum(axis=1)  # the slack balances every snapshot
    print(f'IEEE 14: Bbus {bbus.shape} with {bbus.nnz} nonzeros, Bf {bf.shape}; injections {p.shape}')
    lu = spla.splu(bbus_r)
    many = flows_many(lu, bf_r, p, keep)
    single = flows_one_by_one(bbus_r, bf_r, p, keep)
    base_err = np.abs(many[0] - table['p_pandapower'].to_numpy()).max()
    print(f'max |block solve - separate solves| = {np.abs(many - single).max():.1e} MW; ', end='')
    print(f'snapshot 0 (the base case) matches rundcpp to {base_err:.1e} MW')
    t_many = min(timeit.repeat(lambda: flows_many(spla.splu(bbus_r), bf_r, p, keep), number=1, repeat=5))
    t_single = min(timeit.repeat(lambda: flows_one_by_one(bbus_r, bf_r, p, keep), number=1, repeat=3))
    print(f'{n_timesteps} snapshots: one factorization + block solve {1e3 * t_many:.2f} ms; ', end='')
    print(f'{n_timesteps} separate solves {1e3 * t_single:.1f} ms; ratio {t_single / t_many:.0f}x')


if __name__ == '__main__':
    main()

    Exercise 4¶

    makebdc_shift('case14', 8, 3) (Octave 11.3, MATPOWER 8.1) rebuilds \(A\) from F_BUS and T_BUS, takes \(b = 1/(x\tau)\) with \(\tau = 1\) where TAP is zero, and compares with makeBdc: the largest entry difference of \(B_\text{bus}\) is \(0\). Pfinj of the unmodified case is zero in every entry (no branch has a SHIFT). After setting SHIFT = 3 on branch 8 (the transformer 4–7, \(x = 0.2091\), \(\tau = 0.978\)), exactly one entry of Pfinj is nonzero, \(\text{Pfinj}_8 = -b_8 \cdot 3\pi/180 = -4.8897 \times 0.05236 = -0.25601\) pu (\(-25.6\) MW), and Pbusinj equals \(A^\top\,\text{Pfinj}\) (an injection of \(-0.256\) pu at bus 4 and \(+0.256\) pu at bus 7): both checks hold to \(5.6 \times 10^{-17}\). This is the equivalent-injection formula of the worked example, \(P_{f,\text{shift}} = -\operatorname{diag}(b)\,\theta_\text{shift}\) and \(P_\text{bus,shift} = A^\top P_{f,\text{shift}}\), with MATPOWER's sign: a positive SHIFT reduces the flow from \(f\) to \(t\).

    makebdc_shift.m
    function [err_bbus, pfinj0, pfinj, err_shift] = makebdc_shift(casename, branch_row, shift_deg)
  % Check makeBdc against A' diag(b) A with b = 1/(x tau), and Pfinj against -b * shift (rad), after adding a
  % phase shift to one branch (Chapter 16, Exercise 4). Returns the largest entry differences and both Pfinj.
  define_constants;
  mpc = ext2int(loadcase(casename));
  [Bbus, ~, ~, pfinj0] = makeBdc(mpc);
  nb = size(mpc.bus, 1); nl = size(mpc.branch, 1);
  tau = mpc.branch(:, TAP); tau(tau == 0) = 1;
  b = 1 ./ (mpc.branch(:, BR_X) .* tau);
  A = sparse([1:nl 1:nl], [mpc.branch(:, F_BUS); mpc.branch(:, T_BUS)], [ones(nl, 1); -ones(nl, 1)], nl, nb);
  err_bbus = full(max(max(abs(Bbus - A' * diag(b) * A))));
  mpc.branch(branch_row, SHIFT) = shift_deg;
  [~, ~, Pbusinj, pfinj] = makeBdc(mpc);
  expected = zeros(nl, 1); expected(branch_row) = -b(branch_row) * shift_deg * pi / 180;
  err_shift = max(abs(pfinj - expected));
  err_shift = max(err_shift, full(max(abs(Pbusinj - A' * expected))));   % the bus injections are A' Pfinj
end

    Exercise 5¶

    psdf.ptdf_from_solves builds the PTDF with the chapter's own tool, one DC power flow per column (a unit injected at bus \(i\), withdrawn at the slack), and psdf_matrix applies \(\text{PSDF}_{\ell k} = b_k(\delta_{\ell k} - H_{\ell f_k} + H_{\ell t_k})\). flows_with_shift is the independent check: it models \(\varphi_k\) as the injections \(-b_k\varphi_k\) at \(f_k\), \(+b_k\varphi_k\) at \(t_k\) in a plain DC power flow and adds the direct term \(b_k\varphi_k\) on branch \(k\). On the four-bus ring (\(x = 0.1, 0.2, 0.1, 0.2\), \(P = (0.8, 0.2, -0.5, -0.5)\) pu) every PSDF entry is \(\pm 1/\sum x = \pm 1.6667\) pu per radian: a shift anywhere on a single loop drives the same circulating flow \(\varphi/\sum_\text{loop} x\) around it. A \(5^\circ\) shift on branch 0 moves the flows from \((0.3333, 0.4667, 0.5333, -0.0333)\) to \((0.4788, 0.3212, 0.6788, -0.1788)\) pu, and the PSDF prediction agrees to \(10^{-16}\) (the test repeats this for every branch). Scaled as compute_psdf does, by base_mva and \(\pi/180\), the column reads \(\pm 2.909\) MW per degree.

    The sign convention: ToOp's PSDF is derived for a branch flow \(b_k(\theta_f - \theta_t + \varphi_k)\), so a positive \(\varphi_k\) increases the flow from \(f\) to \(t\). MATPOWER (and PowSyBl's tap changer alpha, which follows the same transformer equation) define the shift the other way, \(p_f = b(\theta_f - \theta_t - \theta_\text{shift})\): a positive shift decreases the from-to flow (Exercise 4 showed \(\text{Pfinj}_8 < 0\) for a positive SHIFT). If the grid model reports its angle in that convention, then \(\varphi_k = -\theta_\text{shift}\), which is exactly the minus sign that get_shift_angles applies in powsybl_backend.py under its TODO. The minus would be unnecessary only if the model's angle already followed the "positive shift increases the from-to flow" convention.

    psdf.py
    """Phase shift distribution factors from DC power flows (Chapter 16, Exercise 5).

ToOp's convention: the flow on branch k is b_k (theta_f - theta_t + phi_k), so the angle phi_k (rad) acts as the
injections -b_k phi_k at f_k and +b_k phi_k at t_k plus a direct flow b_k phi_k on branch k itself, and
PSDF[l, k] = b_k (delta_lk - PTDF[l, f_k] + PTDF[l, t_k]).
"""

import numpy as np
from dc_power_flow import dc_power_flow


def ptdf_from_solves(frm: np.ndarray, to: np.ndarray, x: np.ndarray, n_bus: int, slack: int) -> np.ndarray:
    """PTDF (branches x buses), one DC power flow per column: a unit injected at a bus and withdrawn at the slack."""
    h = np.zeros((len(frm), n_bus))
    for i in range(n_bus):
        if i == slack:
            continue
        p = np.zeros(n_bus)
        p[i], p[slack] = 1.0, -1.0
        h[:, i] = dc_power_flow(frm, to, x, p, slack)
    return h


def psdf_matrix(h: np.ndarray, frm: np.ndarray, to: np.ndarray, x: np.ndarray) -> np.ndarray:
    """PSDF (branches x branches) in flow units per radian of shift, ToOp's sign convention."""
    b = 1 / x
    return b * (np.eye(len(frm)) - h[:, frm] + h[:, to])


def flows_with_shift(
    frm: np.ndarray, to: np.ndarray, x: np.ndarray, p: np.ndarray, slack: int, phi: np.ndarray
) -> np.ndarray:
    """DC flows with shift angles ``phi`` (rad, one per branch) modelled as equivalent injections, ToOp's sign."""
    b = 1 / x
    p_eq = p.copy()
    np.add.at(p_eq, frm, -b * phi)
    np.add.at(p_eq, to, b * phi)
    return dc_power_flow(frm, to, x, p_eq, slack) + b * phi


def main() -> None:
    """Four-bus ring in pu: PSDF column of branch 0 against a DC power flow with a 5-degree shift on that branch."""
    frm, to = np.array([0, 0, 1, 2]), np.array([1, 2, 3, 3])
    x = np.array([0.1, 0.2, 0.1, 0.2])
    p = np.array([0.8, 0.2, -0.5, -0.5])
    base_mva = 100.0
    h = ptdf_from_solves(frm, to, x, 4, slack=0)
    psdf = psdf_matrix(h, frm, to, x)
    print('PSDF (pu per radian), columns = shifted branch; 1 / (sum of x around the ring) = 1 / 0.6:')
    print(np.round(psdf, 4))
    phi = np.zeros(4)
    phi[0] = np.deg2rad(5.0)
    base = dc_power_flow(frm, to, x, p, 0)
    shifted = flows_with_shift(frm, to, x, p, 0, phi)
    print('base flows (pu)       :', np.round(base, 4))
    print('with 5 deg on branch 0:', np.round(shifted, 4))
    print('PSDF prediction       :', np.round(base + psdf[:, 0] * phi[0], 4))
    print(f'max |prediction - power flow| = {np.abs(base + psdf @ phi - shifted).max():.1e}')
    print(
        'MW per degree, as compute_psdf scales it (base_mva * pi / 180):',
        np.round(psdf[:, 0] * base_mva * np.pi / 180, 4),
    )


if __name__ == '__main__':
    main()
