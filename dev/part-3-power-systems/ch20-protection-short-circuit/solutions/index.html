    Chapter 20 – Solutions¶

    Solutions of the Lab and the Exercises of Chapter 20. Try each task before reading its solution. The code is in the chapter's src folder: fault_levels.py (the hand calculation, sequence networks and \(Z_\text{bus}\)), station_sc.py (Lab Part A with pandapower), rx_plane.py (Lab Part B and Exercise 5), case9_sc.py (Exercise 3), switching_sequence.py (Exercise 4) and va_diff_toop.py (Exercise 6, ToOp).     Lab¶

    Part A: short circuit with pandapower¶

    1. The grid. station_sc.station() builds five 220 kV buses: external grids at G1 (s_sc_max_mva=8000, s_sc_min_mva=6000) and G2 (6 000 and 4 500 MVA), both with rx_max=rx_min=0.1; the station busbars S1 and S2 joined by pp.create_switch(net, S1, S2, et="b", type="CB"); lines G1–S1 (40 km), G2–S2 (60 km), S1–L and S2–L (30 km) and G1–G2 (150 km), all \(j0.4\ \Omega/\text{km}\) with \(r = 0.03\ \Omega/\text{km}\) and endtemp_degree=80; a 200 MW, 50 Mvar load at L. The arguments extra_paths=False and r_ohm_per_km=0 reduce it to the grid of the hand calculation.

      station_sc.py
      """Lab Part A of Chapter 20: the worked example's double-busbar station in pandapower, with ``calc_sc`` and ``runpp``.

Two 220 kV infeeds (external grids G1 and G2) reach the station busbars S1 and S2 over 40 km and 60 km of
j0.4 ohm/km line; S1 and S2 are joined by a bus-bus switch of type CB (the coupler). The extra paths the hand
calculation neglects are two 30 km lines from S1 and S2 to a load bus L and a 150 km line G1-G2.
"""

import pandapower as pp
import pandapower.shortcircuit as sc
import pandas as pd

BUSES = ('G1', 'G2', 'S1', 'S2', 'L')


def station(coupler_closed: bool = True, extra_paths: bool = True, r_ohm_per_km: float = 0.03) -> pp.pandapowerNet:
    """Build the station grid; ``extra_paths=False`` and ``r_ohm_per_km=0`` reproduce the hand calculation."""
    net = pp.create_empty_network(sn_mva=100)
    bus = {name: pp.create_bus(net, vn_kv=220, name=name) for name in BUSES}
    pp.create_ext_grid(net, bus['G1'], vm_pu=1.0, s_sc_max_mva=8000, s_sc_min_mva=6000, rx_max=0.1, rx_min=0.1)
    pp.create_ext_grid(net, bus['G2'], vm_pu=1.0, s_sc_max_mva=6000, s_sc_min_mva=4500, rx_max=0.1, rx_min=0.1)

    def line(a: str, b: str, km: float) -> None:
        pp.create_line_from_parameters(
            net,
            bus[a],
            bus[b],
            length_km=km,
            r_ohm_per_km=r_ohm_per_km,
            x_ohm_per_km=0.4,
            c_nf_per_km=0,
            max_i_ka=1.0,
            endtemp_degree=80,  # needed by the minimum case (line resistance at end temperature)
            name=f'{a}-{b}',
        )

    line('G1', 'S1', 40)
    line('G2', 'S2', 60)
    if extra_paths:
        line('S1', 'L', 30)
        line('S2', 'L', 30)
        line('G1', 'G2', 150)
        pp.create_load(net, bus['L'], p_mw=200, q_mvar=50)
    pp.create_switch(net, bus['S1'], bus['S2'], et='b', type='CB', closed=coupler_closed, name='coupler')
    return net


def fault_levels(net: pp.pandapowerNet, case: str = 'max') -> pd.DataFrame:
    """``calc_sc`` with peak current and no branch results; ``ikss_ka``, ``skss_mw`` and ``ip_ka`` per bus."""
    sc.calc_sc(net, case=case, ip=True, branch_results=False)
    table = net.res_bus_sc[['ikss_ka', 'skss_mw', 'ip_ka']].copy()
    table.index = net.bus.name
    return table


def station_table(extra_paths: bool = True, r_ohm_per_km: float = 0.03) -> pd.DataFrame:
    """Steps 2 and 3: I''k at S1 and S2 for both cases, coupler closed and open."""
    rows = {}
    for case in ('max', 'min'):
        for closed in (True, False):
            table = fault_levels(station(closed, extra_paths, r_ohm_per_km), case)
            rows[(case, 'closed' if closed else 'open')] = table.loc[['S1', 'S2'], 'ikss_ka']
    return pd.DataFrame(rows).T


def recoupling_angle() -> tuple[float, float]:
    """Step 4: voltage angle difference S1 - S2 in degrees with the coupler closed and open (AC power flow)."""
    out = []
    for closed in (True, False):
        net = station(closed)
        pp.runpp(net)
        va = net.res_bus.va_degree
        out.append(float(va[net.bus.name == 'S1'].iloc[0] - va[net.bus.name == 'S2'].iloc[0]))
    return out[0], out[1]


def verdict(ik_max_ka: float, ik_min_ka: float, breaker_ka: float, protection_min_ka: float) -> str:
    """Step 5: is a busbar operable with the given maximum and minimum fault currents?"""
    problems = []
    if ik_max_ka > breaker_ka:
        problems.append(f"I''k max {ik_max_ka:.2f} kA exceeds the {breaker_ka:.0f} kA breaker rating")
    if ik_min_ka < protection_min_ka:
        problems.append(f"I''k min {ik_min_ka:.2f} kA is below the {protection_min_ka:.0f} kA the protection needs")
    return 'acceptable' if not problems else '; '.join(problems)


def main() -> None:
    """Print steps 2 to 5 of Lab Part A."""
    pd.set_option('display.width', 120)
    print("full grid (r = 0.03 ohm/km, extra paths), I''k in kA at S1 and S2:")
    print(station_table().round(3))
    for closed in (True, False):
        print('maximum case, coupler', 'closed' if closed else 'open', '(kA, MVA, kA):')
        print(fault_levels(station(closed), 'max').loc[['S1', 'S2']].round(2))
    print('hand-calculation grid (no extra paths, r = 0):')
    print(station_table(extra_paths=False, r_ohm_per_km=0.0).round(3))
    closed, opened = recoupling_angle()
    print(f'angle S1 - S2: {closed:.3f} deg with the coupler closed, {opened:.3f} deg open')
    full = station_table()
    for busbar in ('S1', 'S2'):
        ik_max, ik_min = full.loc[('max', 'open'), busbar], full.loc[('min', 'open'), busbar]
        print(f'split, {busbar}: 8 kA breakers, 5 kA minimum for protection -> {verdict(ik_max, ik_min, 8.0, 5.0)}')
    ik_max, ik_min = full.loc[('max', 'closed'), 'S1'], full.loc[('min', 'closed'), 'S1']
    print(f'coupled: 8 kA breakers, 5 kA minimum -> {verdict(ik_max, ik_min, 8.0, 5.0)}')


if __name__ == '__main__':
    main()

      With the coupler closed both busbars are one node and see the same current; skss_mw is \(\sqrt 3\,U_n I''_k\) (3 962 MVA closed) and ip_ka is \(\kappa\sqrt 2\,I''_k\), about 2.5 times \(I''_k\) here. The coupler lowers the fault level of each busbar once open, by less than the hand calculation predicts: see step 3.

    3. Minimum case and the hand calculation. The minimum case (rows min above) uses \(c_\text{min} = 1.0\), the minimum feeder powers and the line resistances at the end temperature, and is 14–15 % below the maximum. The pandapower values with the coupler open (8.64 and 7.26 kA) are well above the chapter's 6.2 and 4.3 kA because the station is not radial: with the coupler open, busbar S1 is still fed by infeed 2 through S2–L–S1 (60 km) and through G2–G1–S1, and busbar S2 by infeed 1 through the mirror paths. Those parallel paths are what the hand calculation neglects. Removing them (extra_paths=False) and setting \(r = 0\) gives 10.428 kA closed and 6.174 / 4.254 kA open against the chapter's 10.4, 6.2 and 4.3 kA (minimum case 8.942, 5.284 and 3.658 kA). The closed value hardly moves (10.398 against 10.428 kA) because, with both infeeds already on the same node, the extra paths add nothing in parallel that was not there.

    4. The recoupling angle. pp.runpp with the coupler closed gives \(\theta_{S1} - \theta_{S2} = 0\) (one node); open, with the 200 MW load fed from both sides, \(0.359°\) (recoupling_angle()). That is the standing angle a synchro-check relay on the coupler would measure before reclosing: small here, because the two busbars stay tied through the 30 km lines and the load bus.

    5. Verdict. With 8 kA breakers and 5 kA as the smallest current the busbar protection needs (verdict()): coupled, 10.40 kA exceeds the breaker rating, so the station cannot run coupled; split, S2 is acceptable (7.26 kA maximum, 6.31 kA minimum) but S1 still exceeds 8 kA (8.64 kA) because of the parallel paths, so the split alone does not make this station operable, and the operator would also have to open one of the extra paths or accept a breaker upgrade. In the chapter's radial grid the picture is the one the text describes: split, both busbars are below 8 kA, but S2 falls to 4.25 kA (3.66 kA in the minimum case), below the 5 kA the protection needs. Which constraint bites depends on the actual meshing around the station, which is exactly why a flow-only optimizer cannot decide it.

    Part B: distance zones in the R-X plane¶

    1. The polygon. For \(\theta = 30°\), \(r_i = 2.5\), \(r_v = 10\), \(x_v = 15\ \Omega\) the vertices are \((0, 0)\), \((2.5, 0)\), \((2.5, 1.443)\), \((10, 5.774)\), \((10, 15)\) and \((0, 15)\) (rx_plane.toop_polygon): a rectangle \([0, r_v] \times [0, x_v]\) with its lower right corner cut off along the line through the origin at the angle \(\theta\), except for the small foot \(r_i\) along the R axis. In GeoGebra, define the four parameters as sliders and the polygon as Polygon((0,0), (r_i,0), (r_i, r_i tan(θ)), (r_v, r_v tan(θ)), (r_v, x_v), (0, x_v)); in matplotlib, plt.fill(*zip(*poly)) draws the same figure.

    2. The loading path. For a 380 kV line carrying \(S = P + jQ\) the relay sees \(|Z| = V^2/|S|\) at the angle of \(S\) (apparent_impedance): 1 000 MVA at \(\cos\varphi = 0.95\) is \((137.2, 45.1)\ \Omega\), far outside; the point moves towards the origin along the ray at \(\varphi = 18.2°\) as the loading grows and enters the polygon where the ray meets its boundary, at \(|Z| = 2.63\ \Omega\), i.e. at \(S = 380^2 / 2.63 = 54\,872\) MVA (tripping_mva). At unity power factor the boundary is the foot \(r_i = 2.5\ \Omega\) (57 760 MVA); at \(\varphi = 60°\) the sloped edge is met at \(|Z| = 17.3\ \Omega\) (8 337 MVA). Numbers of this size say that the example parameters of the relay table are not primary ohms of a 380 kV line: a 380 kV line loaded to 1 000 MVA sits at \(|Z| = 144\ \Omega\). Secondary ohms (after the CT and VT ratios) or a lower voltage level are where 10 to 15 \(\Omega\) make sense.

    3. The warning factor and absolute values. ToOp divides \(|R|\) and \(|X|\) by the zone factor before the test (trips(poly, z, 1.41)), which is the same as scaling the polygon by 1.41: the tripping MVA at \(\cos\varphi = 0.95\) drops from 54 872 to 38 916 MVA (tripping_mva(..., factor=1.41)). Because of the absolute values the test is non-directional: a reverse flow of 60 000 MVA at \(\cos\varphi = 0.95\) gives \(Z = (-2.29, -0.75)\ \Omega\), a point in the third quadrant, outside the polygon as drawn, yet ToOp's \((|R|, |X|) = (2.29, 0.75)\) is inside and the line trips. A directional relay would not see a forward fault here.

    4. 80° and the validity condition. With \(\theta = 80°\) the third and fourth vertices become \((2.5, 14.18)\) and \((10, 56.71)\): the fourth lies above \((10, 15)\), so the edge from \((10, 56.71)\) to \((10, 15)\) runs downwards and the sloped edge from \((2.5, 14.18)\) to \((10, 56.71)\) crosses the top edge from \((10, 15)\) to \((0, 15)\). is_simple reports the self-intersection that shapely flags as invalid. The polygon is simple exactly when the fourth vertex is not above the fifth, \(r_v \tan\theta \le x_v\), i.e. \(\theta \le \arctan(x_v / r_v) = \arctan 1.5 = 56.3°\) for these parameters (is_valid), together with \(0 \le r_i \le r_v\). The relay table's 80° fails it, the tests' 30° passes.

      rx_plane.py
      """Distance protection in the R-X plane (Chapter 20): ToOp's polygon, apparent impedances and quadrilateral zones.

Lab Part B builds ToOp's "danger" polygon from ``angle``, ``r_i``, ``r_v`` and ``x_v``, tests whether the apparent
impedance of a loaded line falls inside it (with ToOp's warning factor, which divides |R| and |X|), finds the MVA
at which a loading path enters the zone, and derives the validity condition r_v tan(angle) <= x_v. Exercise 5 adds
quadrilateral zones 1 and 2 of a 100 km line and the infeed effect on the apparent impedance. Pure NumPy: the
point-in-polygon and segment-intersection tests are written out, so no geometry library is needed.
"""

import numpy as np

Point = tuple[float, float]


def toop_polygon(angle_deg: float, r_i: float, r_v: float, x_v: float) -> list[Point]:
    """ToOp's ``_build_poly`` vertices: (0,0), (r_i,0), (r_i, r_i tan a), (r_v, r_v tan a), (r_v, x_v), (0, x_v)."""
    t = np.tan(np.radians(angle_deg))
    return [(0.0, 0.0), (r_i, 0.0), (r_i, r_i * t), (r_v, r_v * t), (r_v, x_v), (0.0, x_v)]


def is_valid(angle_deg: float, r_i: float, r_v: float, x_v: float) -> bool:
    """The polygon is simple (no self-intersection) iff r_v tan(angle) <= x_v, with 0 <= r_i <= r_v."""
    return 0.0 <= r_i <= r_v and r_v * np.tan(np.radians(angle_deg)) <= x_v


def _segments_cross(p1: Point, p2: Point, q1: Point, q2: Point) -> bool:
    """Proper intersection of the open segments p1p2 and q1q2 (shared endpoints do not count)."""

    def orient(a: Point, b: Point, c: Point) -> float:
        return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])

    d1, d2 = orient(q1, q2, p1), orient(q1, q2, p2)
    d3, d4 = orient(p1, p2, q1), orient(p1, p2, q2)
    return d1 * d2 < 0 and d3 * d4 < 0


def is_simple(poly: list[Point]) -> bool:
    """True when no two non-adjacent edges cross (what shapely calls a valid polygon)."""
    n = len(poly)
    edges = [(poly[i], poly[(i + 1) % n]) for i in range(n)]
    for i in range(n):
        for j in range(i + 2, n):
            if i == 0 and j == n - 1:
                continue  # adjacent through the closing edge
            if _segments_cross(*edges[i], *edges[j]):
                return False
    return True


def contains(poly: list[Point], point: Point) -> bool:
    """Ray casting: is the point inside the polygon (boundary counts as inside)?"""
    x, y = point
    inside = False
    n = len(poly)
    for i in range(n):
        (x1, y1), (x2, y2) = poly[i], poly[(i + 1) % n]
        if (
            min(y1, y2) <= y <= max(y1, y2)
            and min(x1, x2) <= x <= max(x1, x2)
            and abs((x2 - x1) * (y - y1) - (y2 - y1) * (x - x1)) < 1e-12
        ):
            return True  # on an edge
        if (y1 > y) != (y2 > y):
            x_cross = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < x_cross:
                inside = not inside
    return inside


def apparent_impedance(v_ll_kv: float, p_mw: float, q_mvar: float) -> Point:
    """(R, X) in ohm seen by a relay: |Z| = V_LL^2 / |S| at the angle of P + jQ (ToOp's ``get_complex_impedance``)."""
    s = complex(p_mw, q_mvar)
    if s == 0:
        return (np.inf, np.inf)
    z = v_ll_kv**2 / abs(s) * s / abs(s)
    return (z.real, z.imag)


def trips(poly: list[Point], z: Point, factor: float = 1.0) -> bool:
    """ToOp's test: (|R|, |X|) divided by the zone factor inside the polygon (so the test is non-directional)."""
    return contains(poly, (abs(z[0]) / factor, abs(z[1]) / factor))


def boundary_distance(poly: list[Point], phi_rad: float) -> float:
    """Largest |Z| on the ray from the origin at angle phi that is still inside the polygon (ray-edge intersection)."""
    d = np.array([np.cos(phi_rad), np.sin(phi_rad)])
    best = 0.0
    n = len(poly)
    for i in range(n):
        a, b = np.array(poly[i]), np.array(poly[(i + 1) % n])
        e = b - a
        det = d[0] * (-e[1]) - d[1] * (-e[0])
        if abs(det) < 1e-12:
            continue
        t, u = np.linalg.solve(np.column_stack([d, -e]), a)
        if t >= 0 and -1e-12 <= u <= 1 + 1e-12:
            best = max(best, float(t))
    return best


def tripping_mva(poly: list[Point], v_ll_kv: float, phi_deg: float, factor: float = 1.0) -> float:
    """Smallest |S| along a loading path at power angle phi for which the apparent impedance enters the zone."""
    zb = boundary_distance(poly, np.radians(phi_deg))
    return v_ll_kv**2 / (factor * zb) if zb > 0 else np.inf


def quadrilateral_zone(x_reach_ohm: float, r_reach_ohm: float) -> list[Point]:
    """Exercise 5: a forward quadrilateral zone, reactance reach ``x_reach`` and resistive blinder ``r_reach``."""
    return [(0.0, 0.0), (r_reach_ohm, 0.0), (r_reach_ohm, x_reach_ohm), (0.0, x_reach_ohm)]


def infeed_apparent_impedance(z_line: complex, position: float, r_fault: float, remote_share: float) -> complex:
    """Relay impedance for a fault at ``position`` of the line with fault resistance and a remote infeed.

    With I_remote = remote_share / (1 - remote_share) times the relay current, the fault resistance is seen
    magnified: Z = position Z_line + R_f (I_relay + I_remote) / I_relay.
    """
    magnification = 1.0 / (1.0 - remote_share)
    return position * z_line + r_fault * magnification

    )

    Exercise 1¶

    For a fault on phase a through zero impedance, the boundary conditions are \(I_b = I_c = 0\) and \(V_a = 0\). Transforming the currents, \(I_0 = I_1 = I_2 = \tfrac13 (I_a + 0 + 0) = I_a / 3\): the three sequence currents are equal. The sequence voltages at the fault are \(V_1 = E - Z_1 I_1\), \(V_2 = -Z_2 I_2\), \(V_0 = -Z_0 I_0\), and \(V_a = V_0 + V_1 + V_2 = 0\) gives \(E - (Z_1 + Z_2 + Z_0)\,I_1 = 0\), so

    \[ I_1 = \frac{E}{Z_1 + Z_2 + Z_0}, \qquad I_f = I_a = 3 I_1 = \frac{3E}{Z_1 + Z_2 + Z_0}, \]

    the three sequence networks in series. The three-phase fault is \(I_f^{(3)} = E / Z_1\), so

    \[ \frac{I_f^{(1)}}{I_f^{(3)}} = \frac{3 Z_1}{Z_1 + Z_2 + Z_0} \approx \frac{3 Z_1}{2 Z_1 + Z_0} \quad (Z_2 \approx Z_1), \]

    which exceeds one exactly when \(Z_0 < Z_1\) (fault_levels.slg_current: 1.2 for \(Z_0 = Z_1/2\), 1.0 for \(Z_0 = Z_1\), 0.6 for \(Z_0 = 3 Z_1\)). Lines have \(Z_0 \approx 3 Z_1\), so far from the sources the single-phase current is the smaller one. At a solidly grounded transformer the zero-sequence path is short: a Yg–Δ winding lets zero-sequence current circulate in the delta and presents only its leakage impedance to ground, and a generator's \(Z_0\) is also small, so near such transformers \(Z_0 < Z_1\) and the single line-to-ground current exceeds the three-phase one. That is why breaker ratings and earthing designs are checked against the single-phase fault as well, and why a busbar split that changes \(Z_1\) at a station changes the two fault types by different amounts.

    fault_levels.py
    """Fault levels by hand (Chapter 20): the IEC 60909 equivalent voltage source, Z_bus and a busbar split.

The worked example of the chapter (two infeeds on a double busbar, coupler closed or open), the single
line-to-ground current of Exercise 1 and the five-bus Z_bus calculation of Exercise 2, with a split node. All
impedances are in ohm at the fault voltage level, currents in kA, and the voltage factor c multiplies the
nominal line-to-line voltage: I''k = c Un / (sqrt(3) |Zk|).
"""

from __future__ import annotations  # ToOp's interpreter is older than 3.14: ``split`` returns its own class

from dataclasses import dataclass

import numpy as np


def feeder_impedance(skq_mva: float, un_kv: float = 220.0, c: float = 1.1) -> float:
    """Network feeder impedance Z_Q = c Un^2 / S''kQ in ohm."""
    return c * un_kv**2 / skq_mva


def ikss_ka(zk_ohm: float, un_kv: float = 220.0, c: float = 1.1) -> float:
    """Initial symmetrical short-circuit current c Un / (sqrt(3) Zk) in kA."""
    return c * un_kv / (np.sqrt(3) * zk_ohm)


def skss_mva(ik_ka: float, un_kv: float = 220.0) -> float:
    """Short-circuit power sqrt(3) Un I''k in MVA."""
    return np.sqrt(3) * un_kv * ik_ka


def worked_example(skq=(8000.0, 6000.0), km=(40.0, 60.0), x_per_km=0.4, un_kv=220.0, c=1.1) -> dict[str, float]:
    """The chapter's station: path impedances, the parallel fault level and the two split fault levels."""
    zq = [feeder_impedance(s, un_kv, c) for s in skq]
    z = [q + x_per_km * d for q, d in zip(zq, km, strict=True)]
    zk = 1.0 / sum(1.0 / zi for zi in z)
    return {
        'zq1_ohm': zq[0],
        'zq2_ohm': zq[1],
        'z1_ohm': z[0],
        'z2_ohm': z[1],
        'zk_closed_ohm': zk,
        'ik_closed_ka': ikss_ka(zk, un_kv, c),
        'ik_open_busbar1_ka': ikss_ka(z[0], un_kv, c),
        'ik_open_busbar2_ka': ikss_ka(z[1], un_kv, c),
    }


def three_phase_current(e: complex, z1: complex) -> complex:
    """Three-phase fault current E / Z1 (positive sequence only)."""
    return e / z1


def slg_current(e: complex, z1: complex, z2: complex, z0: complex) -> complex:
    """Single line-to-ground fault current 3 E / (Z1 + Z2 + Z0): the three sequence networks in series."""
    return 3 * e / (z1 + z2 + z0)


@dataclass(frozen=True)
class Grid:
    """A small grid for Z_bus: branches (from, to, x ohm) and source impedances to ground per bus (ohm, 0 = none)."""

    branches: list[tuple[int, int, float]]
    sources: dict[int, float]
    n_bus: int

    def ybus(self) -> np.ndarray:
        """Nodal admittance matrix in siemens (reactances only, so it is real and the j is left implicit)."""
        y = np.zeros((self.n_bus, self.n_bus))
        for i, j, x in self.branches:
            y[i, i] += 1 / x
            y[j, j] += 1 / x
            y[i, j] -= 1 / x
            y[j, i] -= 1 / x
        for i, x in self.sources.items():
            y[i, i] += 1 / x
        return y

    def zbus(self) -> np.ndarray:
        """Z_bus = Y_bus^-1; its diagonal is the Thevenin impedance at each bus."""
        return np.linalg.inv(self.ybus())

    def fault_currents(self, un_kv: float, c: float = 1.1) -> np.ndarray:
        """I''k at every bus in kA, from the diagonal of Z_bus."""
        return np.array([ikss_ka(z, un_kv, c) for z in np.diag(self.zbus())])

    def split(self, bus: int, moved: list[int]) -> Grid:
        """Split ``bus``: branches with the given indices move their ``bus`` end to a new node, as in Chapter 18."""
        new = self.n_bus
        branches = []
        for k, (i, j, x) in enumerate(self.branches):
            if k in moved:
                i, j = (new if i == bus else i), (new if j == bus else j)
            branches.append((i, j, x))
        return Grid(branches=branches, sources=dict(self.sources), n_bus=self.n_bus + 1)


# Exercise 2: five 110 kV buses, two infeeds (buses 0 and 4), a meshed middle; reactances in ohm.
FIVE_BUS = Grid(

def split_comparison(grid: Grid = FIVE_BUS, bus: int = 2, moved: list[int] | None = None, un_kv: float = 110.0):
    """Exercise 2: fault currents before and after moving two branches of ``bus`` to a new node, and the changes."""
    moved = [2, 5] if moved is None else moved  # branches 2-3 and 0-2 leave bus 2 for the new node
    before = grid.fault_currents(un_kv)
    after = grid.split(bus, moved).fault_currents(un_kv)
    change = 100 * (after[: grid.n_bus] - before) / before
    return before, after, change

    fault_levels.FIVE_BUS is a 110 kV grid with reactances in ohm: branches 1–2 (8), 2–3 (6), 3–4 (6), 4–5 (8), 2–4 (10) and 1–3 (12), and source impedances of 2 \(\Omega\) at bus 1 and 3 \(\Omega\) at bus 5 (1-based bus numbers). \(Y_\text{bus}\) holds the branch admittances plus the source admittances on the diagonal, and \(Z_\text{bus} = Y_\text{bus}^{-1}\); the diagonal entry \(Z_{kk}\) is the Thevenin impedance at bus \(k\) and \(I''_k = 1.1 \cdot 110 / (\sqrt 3\, Z_{kk})\). Before the split: 38.47, 12.84, 12.37, 12.86 and 27.02 kA at buses 1 to 5. Splitting bus 3 as in Chapter 18, branches 3–4 and 1–3 moving to a new node 6 and branch 2–3 staying: 38.42, 10.61, 5.55, 12.70, 26.96 kA and 9.85 kA at the new node. Changes: −0.1 %, −17.4 %, −55.1 %, −1.2 %, −0.2 %: buses 2 and 3 change by more than 10 % (split_comparison). Bus 3 loses the parallel contribution of infeed 1 through 1–3 (its half of the old station now hangs on 2–3 alone), and bus 2 loses the path 2–3–4 towards infeed 2. Buses 1 and 5 barely move because their \(Z_{kk}\) is dominated by the source impedance right at the bus (2 and 3 \(\Omega\)) and the network beyond contributes only a parallel branch of tens of ohms; a change deep in the mesh alters that parallel branch by a little, and a little of a small contribution is nothing. In \(Z_\text{bus}\) terms: a split is a low-rank change of \(Y_\text{bus}\), and the resulting change of \(Z_{kk}\) is proportional to the square of the transfer impedance between bus \(k\) and the split station, which decays with electrical distance. Fault levels are a local property, which is why a split to limit them works, and why it does not help a remote station.

    case9_sc.py
    """Exercise 3 of Chapter 20: maximum and minimum short-circuit currents of pandapower's ``case9``.

``case9`` ships without short-circuit data. ``with_sc_data`` adds what ``calc_sc`` needs: the external grid's
``s_sc_max_mva``, ``s_sc_min_mva``, ``rx_max`` and ``rx_min``; the generators' ``vn_kv``, ``sn_mva``, ``xdss_pu``,
``rdss_ohm`` and ``cos_phi``; and the lines' ``endtemp_degree`` for the minimum case. The voltage factor is c_max
= 1.1 for the maximum case and c_min = 1.0 for the minimum case at this voltage level (IEC 60909 table 1).
"""

import pandapower as pp
import pandapower.networks as pn
import pandapower.shortcircuit as sc
import pandas as pd


def with_sc_data(
    s_sc_max_mva: float = 5000.0,
    s_sc_min_mva: float = 3000.0,
    gen_sn_mva: float = 250.0,
    xdss_pu: float = 0.2,
    endtemp_degree: float = 80.0,
) -> pp.pandapowerNet:
    """``case9`` with the short-circuit parameters of the external grid, the generators and the lines."""
    net = pn.case9()
    net.ext_grid['s_sc_max_mva'] = s_sc_max_mva
    net.ext_grid['s_sc_min_mva'] = s_sc_min_mva
    net.ext_grid['rx_max'] = 0.1
    net.ext_grid['rx_min'] = 0.1
    net.gen['vn_kv'] = net.bus.vn_kv[net.gen.bus].to_numpy()
    net.gen['sn_mva'] = gen_sn_mva
    net.gen['xdss_pu'] = xdss_pu
    net.gen['rdss_ohm'] = 0.05
    net.gen['cos_phi'] = 0.85
    net.line['endtemp_degree'] = endtemp_degree
    return net


def max_min_table(net: pp.pandapowerNet | None = None) -> pd.DataFrame:
    """``ikss_ka`` for ``case="max"`` and ``case="min"`` at every bus, and the ratio min/max."""
    net = with_sc_data() if net is None else net
    out = {}
    for case in ('max', 'min'):
        sc.calc_sc(net, case=case, ip=True, branch_results=False)
        out[case] = net.res_bus_sc.ikss_ka.copy()
    table = pd.DataFrame(out)
    table['min/max'] = table['min'] / table['max']
    return table


def endtemp_effect(temps=(20.0, 80.0, 160.0)) -> pd.Series:
    """Minimum-case I''k at bus 5 (0-based 4, in the middle of a long line) for several line end temperatures."""
    out = {}
    for t in temps:
        net = with_sc_data(endtemp_degree=t)
        sc.calc_sc(net, case='min', branch_results=False)
        out[t] = float(net.res_bus_sc.ikss_ka[4])
    return pd.Series(out, name='ikss_min_bus5_ka')


def main() -> None:
    """Print the comparison and the role of the end temperature."""
    table = max_min_table()
    print("case9 (345 kV): I''k in kA, max (c = 1.1, S''kQ 5000 MVA) and min (c = 1.0, S''kQ 3000 MVA, hot lines)")
    print(table.round(3).to_string())
    print(
        'ratio min/max: smallest',
        round(table['min/max'].min(), 3),
        'at bus',
        int(table['min/max'].idxmin()),
        '; largest',
        round(table['min/max'].max(), 3),
        'at bus',
        int(table['min/max'].idxmax()),
    )
    print(
        'minimum case at bus 5 for line end temperatures 20, 80, 160 degC (kA):', endtemp_effect().round(4).to_dict()
    )


if __name__ == '__main__':
    main()

    switching_sequence.py
    """Exercise 4 of Chapter 20: a switching sequence in a double-busbar, single-breaker station, checked step by step.

The station is a graph of nodes (busbars BB1 and BB2, the line bay's node between its breaker and its busbar
disconnectors, the coupler bay's node, the line) joined by switching devices. A breaker may switch under load. A
disconnector may only operate when no current flows through it, that is when its two ends are already joined by
another closed path (the voltage across it is zero and the current commutates), or when one end is dead (not
connected to any source or load). ``run`` applies a sequence and raises on the first forbidden operation; it also
reports the moments at which the line is in parallel across both busbars.
"""

from __future__ import annotations  # ToOp's interpreter is older than 3.14: ``double_busbar`` returns its class

from dataclasses import dataclass, field

import networkx as nx


@dataclass
class Device:
    """A switching device between two nodes; ``breaker`` tells whether it may switch under load."""

    a: str
    b: str
    breaker: bool
    closed: bool


@dataclass
class Station:
    """A double-busbar station with one line bay and one coupler bay; ``sources`` are the live nodes."""

    devices: dict[str, Device] = field(default_factory=dict)
    sources: set[str] = field(default_factory=set)

    @classmethod
    def double_busbar(cls) -> Station:
        """Line bay on busbar 1, coupler open, both busbars fed by other bays (modelled as live nodes)."""
        station = cls(sources={'BB1', 'BB2', 'LINE'})
        station.devices = {
            'Q1': Device('BAY', 'BB1', breaker=False, closed=True),  # busbar disconnector to BB1
            'Q2': Device('BAY', 'BB2', breaker=False, closed=False),  # busbar disconnector to BB2
            'CB': Device('BAY', 'LINE_SIDE', breaker=True, closed=True),  # line bay breaker
            'Q9': Device('LINE_SIDE', 'LINE', breaker=False, closed=True),  # line disconnector
            'QC1': Device('COUPLER', 'BB1', breaker=False, closed=False),  # coupler disconnectors
            'QC2': Device('COUPLER', 'BB2', breaker=False, closed=False),
            'CBC': Device('COUPLER', 'COUPLER_B', breaker=True, closed=False),  # coupler breaker
            'QC3': Device('COUPLER_B', 'BB2', breaker=False, closed=False),  # ToOp-style coupler bay: CB between
        }
        # a coupler bay is BB1 - QC1 - COUPLER - CBC - COUPLER_B - QC3 - BB2; QC2 is not used by the sequence
        del station.devices['QC2']
        return station

    def graph(self, without: str | None = None) -> nx.Graph:
        """The nodes joined by closed devices, optionally ignoring one device."""
        g = nx.Graph()
        for name, d in self.devices.items():
            g.add_nodes_from([d.a, d.b])
            if d.closed and name != without:
                g.add_edge(d.a, d.b)
        return g

    def live(self, node: str, g: nx.Graph) -> bool:
        """A node is live when a closed path connects it to a source."""
        return any(nx.has_path(g, node, s) for s in self.sources if s in g)

    def can_operate(self, name: str) -> tuple[bool, str]:
        """Whether the device may change state now, and why not."""
        d = self.devices[name]
        if d.breaker:
            return True, 'breaker'
        g = self.graph(without=name)
        if nx.has_path(g, d.a, d.b):
            return True, 'parallel path, no voltage across the disconnector'
        if not (self.live(d.a, g) and self.live(d.b, g)):
            return True, 'one side dead, no current'
        return False, 'would make or break load current'

    def line_on_both_busbars(self) -> bool:
        """Is the line connected to BB1 and BB2 at the same time (through the bay or the coupler)?"""
        g = self.graph()
        return nx.has_path(g, 'LINE', 'BB1') and nx.has_path(g, 'LINE', 'BB2')

    def busbars_coupled(self) -> bool:
        """Are BB1 and BB2 electrically one?"""
        return nx.has_path(self.graph(), 'BB1', 'BB2')


# Move the line bay from BB1 to BB2, then split the station (coupler open).
SEQUENCE = [
    ('close', 'QC1', 'coupler disconnector to BB1'),
    ('close', 'QC3', 'coupler disconnector to BB2'),
    ('close', 'CBC', 'coupler breaker: BB1 and BB2 are now paralleled'),
    ('close', 'Q2', 'bay disconnector to BB2 (make before break: Q1 still closed)'),
    ('open', 'Q1', 'bay disconnector to BB1 (its current commutated through Q2 and the coupler)'),
    ('open', 'CBC', 'coupler breaker: the station is split'),
    ('open', 'QC1', 'coupler disconnectors, dead'),
    ('open', 'QC3', ''),
]


def run(station: Station, sequence=SEQUENCE) -> list[dict[str, object]]:
    """Apply the sequence; raise ValueError on a forbidden operation; return one record per step."""
    log = []
    for action, name, note in sequence:
        ok, why = station.can_operate(name)
        if not ok:
            raise ValueError(f'{action} {name}: {why}')
        station.devices[name].closed = action == 'close'
        log.append(
            {
                'step': f'{action} {name}',
                'device': 'breaker' if station.devices[name].breaker else 'disconnector',
                'allowed because': why,
                'line on both busbars': station.line_on_both_busbars(),
                'busbars coupled': station.busbars_coupled(),
                'note': note,
            }
        )
    return log


def main() -> None:
    """Print the sequence with the checks, then show that opening Q1 without the coupler is refused."""
    for row in run(Station.double_busbar()):
        print(f'{row["step"]:10s} {row["device"]:12s} parallel={str(row["line on both busbars"]):5s}', end=' ')
        print(f'coupled={str(row["busbars coupled"]):5s} {row["allowed because"]}; {row["note"]}')
    try:
        run(Station.double_busbar(), [('close', 'Q2', ''), ('open', 'Q1', '')])
    except ValueError as e:
        print('wrong order, coupler open:', e)
    try:
        run(Station.double_busbar(), [('open', 'Q1', '')])
    except ValueError as e:
        print('opening Q1 under load:', e)


if __name__ == '__main__':
    main()

    Exercise 5¶

    The line is \(Z_L = 100 \cdot (0.03 + j0.3) = 3 + j30\ \Omega\). Zone 1 reaches 80 %, \(X = 24\ \Omega\), zone 2 120 %, \(X = 36\ \Omega\); both quadrilaterals are drawn with a resistive blinder at \(R = 10\ \Omega\) and the reactance line as their top edge (rx_plane.quadrilateral_zone; in GeoGebra, Polygon((0,0), (10,0), (10,24), (0,24)) and the same with 36, plus the line impedance as a segment from the origin to \((3, 30)\)). A fault at 90 % of the line with \(R_f = 5\ \Omega\) seen from the near end, with all the fault current supplied through the relay, is \(Z = 0.9\,Z_L + R_f = (7.7, 27.0)\ \Omega\): beyond zone 1, inside zone 2, as it should be for a fault in the last 20 % of the line. Now let the split busbar at the remote end feed the fault as well, with half the fault current: the relay's current is half the fault current, so the fault resistance appears doubled, \(Z = 0.9\,Z_L + 2 R_f = (12.7, 27.0)\ \Omega\) (infeed_apparent_impedance). The point moves to the right by \(5\ \Omega\), past the resistive blinder of both zones: the relay under-reaches, it sees an impedance larger than the fault position, and neither zone 1 nor zone 2 picks up; clearing is left to the backup zone or the remote end. Infeed always makes a distance relay under-reach (the apparent impedance grows); zone 1 is set to 80 % precisely so that errors of this kind do not make it over-reach into the next line. A topology change that adds an infeed behind a relay therefore changes the apparent fault position, and settings validated for the unsplit station may not protect the split one.

    Exercise 6¶

    va_diff_toop.py runs the Chapter 19 AC N-1 (nine single-line outages on grid.xiidm) through the Chapter 19 module and reads va_diff_results: nine rows, one per outaged branch, each with the angle across the open ends of the outaged line (the grid has no open retained switches, so no switch rows). None exceeds 20° or 10°; the largest are 0.485° for the outages of L1 and L2, 0.424° for L8 and L3, 0.179° for L4 to L7, and the L9 row is NaN because that outage leaves one end without voltage (no angle to compare). The worst case, reclosing L1 after its outage: an operator (or the automatic reclosing scheme) would check that the fault is cleared and the protection has reset, that both ends are live and within their voltage bands, that the standing angle across the open breaker (0.49° here, far below a typical synchro-check setting of 10–20° and below ToOp's critical_va_diff_degree = 20) and the slip are within limits, and then close one end first and the other with synchro-check supervision. L9 is different: with one end dead there is nothing to synchronise; it is energised from the live end ("dead line, live bus" closing) and the remote breaker is closed afterwards under synchro-check. The metric does its job: a large standing angle after a topology change is what would turn a routine reclosing into a torque shock on nearby machines, and that is what max_va_diff_n_1 flags.

    va_diff_toop.py
    """Exercise 6 of Chapter 20: the angle differences of the Chapter 19 AC N-1, above 20 and 10 degrees.

Run it with ToOp's interpreter from this folder; it reuses the Chapter 19 module ``ac_n1_toop.py`` (found two
folders up, in ``ch19-operational-security/src``) to run the AC N-1 of ``data/grid_node_breaker/grid.xiidm``, then
lists the ``va_diff_results`` rows above the two thresholds and the worst one. ``above`` and ``worst`` are plain
pandas and need no ToOp; ``main`` does.
"""

from __future__ import annotations

import sys
from pathlib import Path
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    import pandas as pd

CH19_SRC = Path(__file__).resolve().parents[2] / 'ch19-operational-security' / 'src'
THRESHOLDS = (20.0, 10.0)


def above(va_diff: pd.DataFrame, degrees: float) -> pd.DataFrame:
    """The rows whose |va_diff| exceeds ``degrees``, worst first (NaN rows, an islanded end, are dropped)."""
    rows = va_diff[va_diff.va_diff.abs() > degrees]
    return rows.reindex(rows.va_diff.abs().sort_values(ascending=False).index)


def worst(va_diff: pd.DataFrame) -> pd.Series:
    """The row with the largest |va_diff|."""
    return va_diff.loc[va_diff.va_diff.abs().idxmax()]


def main() -> None:
    """Run the Chapter 19 AC N-1 and report the angle differences."""
    sys.path.insert(0, str(CH19_SRC))
    import ac_n1_toop
    import pypowsybl

    net = pypowsybl.network.load(str(ac_n1_toop.default_grid_path()))
    results = ac_n1_toop.run(net, ac_n1_toop.n1_definition(net))
    va = ac_n1_toop.frames(results)['va_diff']
    print(
        f'{len(va)} va_diff rows (one per outaged branch), {int(va.va_diff.isna().sum())} NaN',
        '(an end left without voltage)',
    )
    for degrees in THRESHOLDS:
        rows = above(va, degrees)
        print(f'rows above {degrees:.0f} deg: {len(rows)}')
    w = worst(va)
    print(f'worst: contingency {w.contingency}, element {w.element}, va_diff {w.va_diff:.3f} deg')
    print(
        va[['contingency', 'element', 'va_diff']]
        .sort_values('va_diff', key=abs, ascending=False)
        .to_string(index=False)
    )


if __name__ == '__main__':
    main()
