def reset_day(env, chronic: int, seed: int):
    """Start ``chronic`` at its first step with a horizon of one day, seeded (only the random agent uses the seed)."""
    env.seed(seed)
    return env.reset(seed=seed, options={'time serie id': chronic, 'max step': N_STEPS})


def shaped_reward(obs, changed: bool, done: bool, step: int) -> float:
    """The reward of ``grid_env.py``: minus the overload, minus 0.02 per change, minus 5 for an early game over."""
    reward = -float(np.maximum(0.0, obs.rho - 1.0).sum()) - 0.02 * changed
    if done and step < N_STEPS:
        reward -= 5.0
    return reward

      The stand-in grid_env.py of the fallback track has 5 buses, 6 lines of 15 to 130 MW, a slack generator at bus 1 and loads at buses 3, 4 and 5 (35 %, 40 %, 25 % of a daily profile with a 150 MW evening peak, 4 % noise per step); the observation is the vector rho of the six loadings, the topology is one of 7 (every line closed, or one line open: none of the six is a bridge), and the 8 actions are "do nothing" and "switch to topology \(t\)". At the peak the meshed topology loads the weak tie line 2–3 (15 MW) to 114 %, two switches relieve it (open 2–3: 70 %; open 4–5: 80 %) and the other four are destructive (250 % to 583 %), which is the shape of the real action space: most topology actions are harmful. A line above 100 % for more than 2 consecutive steps (Grid2Op's soft-overflow allowance) or above 150 % trips and ends the episode, with a reward of −5.

      grid_env.py (environment)
      def load_profile(step: np.ndarray) -> np.ndarray:
    """Total load in MW: a morning ramp and an evening peak on a day of ``N_STEPS`` steps."""
    t = step / N_STEPS
    return PEAK_MW * (0.45 + 0.25 * np.exp(-(((t - 0.35) / 0.08) ** 2)) + 0.55 * np.exp(-(((t - 0.78) / 0.07) ** 2)))


def topologies() -> list[np.ndarray]:
    """Action set: every line in service, then one line open at a time (bridges excluded, they island the grid)."""
    out = [np.ones(len(LINES), dtype=bool)]
    for k in range(len(LINES)):
        mask = np.ones(len(LINES), dtype=bool)
        mask[k] = False
        if ptdf(mask) is not None:
            out.append(mask)
    return out


def ptdf(in_service: np.ndarray) -> np.ndarray | None:
    """PTDF of the lines in service with bus 0 as slack, or ``None`` when the grid is islanded."""
    lines, n_bus = LINES[in_service], len(LOAD_SHARE)
    a = np.zeros((len(lines), n_bus))
    a[np.arange(len(lines)), lines[:, 0]] = 1
    a[np.arange(len(lines)), lines[:, 1]] = -1
    adjacency = np.zeros((n_bus, n_bus))
    adjacency[lines[:, 0], lines[:, 1]] = 1
    if connected_components(adjacency, directed=False)[0] > 1:
        return None
    b = 1 / X[in_service]
    bbus = a.T @ (b[:, None] * a)
    h = np.zeros((len(lines), n_bus))
    h[:, 1:] = (b[:, None] * a[:, 1:]) @ np.linalg.inv(bbus[1:, 1:])
    return h


TOPOLOGIES = topologies()
PTDFS = [ptdf(mask) for mask in TOPOLOGIES]


def rho(topology: int, load_mw: float) -> np.ndarray:
    """Loading of every line (0 for an open line) when the total load is ``load_mw``, split by ``LOAD_SHARE``."""
    mask = TOPOLOGIES[topology]
    p = -LOAD_SHARE * load_mw
    p[0] = -p[1:].sum()
    out = np.zeros(len(LINES))
    out[mask] = np.abs(PTDFS[topology] @ p) / RATING[mask]
    return out


class GridEnv:
    """One episode: the state is (step, topology); ``reset`` draws the day's load noise from the seed."""

    n_actions = len(TOPOLOGIES) + 1  # 0 = do nothing, k = switch to topology k - 1

    def __init__(self, seed: int, noise: float = 0.04):
        """``seed`` drives the day's load noise (relative standard deviation ``noise``) and the random agent."""
        self.rng = np.random.default_rng(seed)
        self.noise = noise

    def reset(self) -> tuple[np.ndarray, int]:
        """Start a day; returns the observation (rho) and the topology index."""
        self.step_index = 0
        self.topology = 0
        self.overflow = np.zeros(len(LINES), dtype=int)  # consecutive steps above 100 % per line
        self.load = load_profile(np.arange(N_STEPS)) * (1 + self.noise * self.rng.standard_normal(N_STEPS))
        return rho(0, self.load[0]), 0

    def simulate(self, action: int) -> np.ndarray:
        """What ``obs.simulate`` does in Grid2Op: the loading after ``action`` at the next step, no state change."""
        topology = self.topology if action == 0 else action - 1
        return rho(topology, self.load[min(self.step_index + 1, N_STEPS - 1)])

    def step(self, action: int) -> tuple[np.ndarray, float, bool]:
        """Apply the action and advance one step: reward = -overload - switching cost; done at a trip or day end."""
        changed = action != 0 and action - 1 != self.topology
        if action != 0:
            self.topology = action - 1
        self.step_index += 1
        done = self.step_index >= N_STEPS - 1
        loading = rho(self.topology, self.load[self.step_index])
        overload = np.maximum(0, loading - 1).sum()
        reward = -overload - 0.02 * changed
        self.overflow = np.where(loading > 1, self.overflow + 1, 0)
        if loading.max() > CASCADE_RHO or self.overflow.max() > OVERFLOW_STEPS:
            reward -= 5.0  # a line trips: game over, as in Grid2Op
            done = True
        return loading, reward, done

      class Agents:
    """The lab's agents over the action list ``actions`` (index 0 is do-nothing, the rest unitary topologies)."""

    def __init__(self, env, actions: list, seed: int = 0):
        """``actions[0]`` must be the do-nothing action; ``seed`` drives the random agent."""
        self.env = env
        self.actions = actions
        self.rng = np.random.default_rng(seed)
        self.simulations = 0  # power flows spent by the greedy agent, to compare with the learned agents

    def do_nothing(self, obs) -> int:
        """Never acts."""
        return 0

    def random(self, obs) -> int:
        """Plays a random topology action when a line passes the threshold."""
        return int(self.rng.integers(1, len(self.actions))) if obs.rho.max() > THRESHOLD else 0

    def greedy(self, obs) -> int:
        """Simulates every action when a line passes the threshold and plays the one with the lowest maximum loading.

        A simulation that ends the game (``done``) or is illegal counts as infinitely bad; do-nothing is candidate 0,
        so the agent acts only when some topology beats staying put.
        """
        if obs.rho.max() <= THRESHOLD:
            return 0
        worst = np.full(len(self.actions), np.inf)
        for i, action in enumerate(self.actions):
            sim_obs, _, sim_done, sim_info = obs.simulate(action)
            self.simulations += 1
            if not sim_done and not sim_info['is_illegal'] and not sim_info['is_ambiguous']:
                worst[i] = sim_obs.rho.max()
        return int(np.argmin(worst))


def run_episode(env, agent, chronic: int, seed: int, actions: list, trace: list | None = None) -> dict:
    """One day of ``chronic`` played by ``agent(obs) -> action index``: survival, mean max loading, reward, ms/step."""
    obs = reset_day(env, chronic, seed)
    total, maxima, done, step = 0.0, [], False, 0
    t0 = time.perf_counter()
    while not done:
        index = agent(obs)
        obs, _, done, info = env.step(actions[index])
        step += 1
        changed = index != 0 and not info['is_illegal']
        total += shaped_reward(obs, changed, done, step)
        maxima.append(float(obs.rho.max()))
        if trace is not None:
            trace.append(index)
    return {
        'survival': step,
        'mean_max_rho': float(np.mean(maxima)),
        'reward': total,
        'ms_per_step': 1e3 * (time.perf_counter() - t0) / step,
    }


def state_index(obs, in_reference: bool) -> int:
    """Discrete state: bin of the maximum loading, most loaded line, whether the grid is in its reference topology."""
    bin_index = int(np.searchsorted(RHO_BINS, obs.rho.max()))
    return (bin_index * len(obs.rho) + int(obs.rho.argmax())) * 2 + int(in_reference)


def n_states(env) -> int:
    """Size of the Q table's first axis."""
    return (len(RHO_BINS) + 1) * env.n_line * 2


def frequent_actions(env, unitary: list, chronics=TRAIN_CHRONICS, n_top: int = 20) -> tuple[list[int], dict]:
    """Lab step 3: the ``n_top`` unitary actions the greedy agent plays most often on the training chronics.

    Returns the indices into ``unitary`` (most frequent first), the play counts and the greedy agent's episode
    statistics on those days (with the number of simulations it spent).
    """
    actions = [env.action_space({}), *unitary]
    counts = np.zeros(len(actions), dtype=int)
    stats = []
    for chronic in chronics:
        agents = Agents(env, actions, seed=chronic)
        trace: list[int] = []
        stats.append(run_episode(env, agents.greedy, chronic, chronic, actions, trace))
        stats[-1]['simulations'] = agents.simulations
        counts += np.bincount(trace, minlength=len(actions))
    counts[0] = 0
    top = [int(i) - 1 for i in np.argsort(-counts, kind='stable')[:n_top] if counts[i] > 0]
    return top, {'counts': counts[1:], 'stats': stats}


def q_learning(
    env,
    actions: list,
    chronics=TRAIN_CHRONICS,
    episodes: int = 200,
    seed: int = 100,
    alpha: float = 0.2,
    gamma: float = 0.95,
    eps: float = 0.3,
) -> np.ndarray:
    """Tabular Q-learning over ``actions`` on the days ``chronics`` in turn; epsilon decays linearly to 0.01."""
    rng = np.random.default_rng(seed)
    q = np.zeros((n_states(env), len(actions)))
    for episode in range(episodes):
        epsilon = max(0.01, eps * (1 - episode / episodes))
        chronic = int(chronics[episode % len(chronics)])
        obs = reset_day(env, chronic, seed + episode)
        in_reference, done, step = True, False, 0
        s = state_index(obs, in_reference)
        while not done:
            i = int(rng.integers(len(actions))) if rng.random() < epsilon else int(np.argmax(q[s]))
            obs, _, done, info = env.step(actions[i])
            step += 1
            changed = i != 0 and not info['is_illegal']
            in_reference = in_reference and not changed
            reward = shaped_reward(obs, changed, done, step)
            s_next = state_index(obs, in_reference)
            target = reward if (done and step < N_STEPS) else reward + gamma * q[s_next].max()
            q[s, i] += alpha * (target - q[s, i])
            s = s_next
    return q


class QPolicy:
    """The greedy policy of a Q table in the signature of the other agents (tracks the reference-topology flag)."""

    def __init__(self, q: np.ndarray):
        """A fresh policy starts in the reference topology."""
        self.q = q
        self.in_reference = True

    def __call__(self, obs) -> int:
        """The action with the highest Q value in the observed state."""
        i = int(np.argmax(self.q[state_index(obs, self.in_reference)]))
        if i != 0:
            self.in_reference = False
        return i


def compare(env, actions: list, chronics=CHRONICS) -> dict:
    """Do-nothing, random and greedy agents over ``actions`` on the same chronics and seeds."""
    out: dict[str, list[dict]] = {'do-nothing': [], 'random': [], 'greedy (simulate)': []}
    for chronic in chronics:
        agents = Agents(env, actions, seed=chronic)
        out['do-nothing'].append(run_episode(env, agents.do_nothing, chronic, chronic, actions))
        out['random'].append(run_episode(env, agents.random, chronic, chronic, actions))
        out['greedy (simulate)'].append(run_episode(env, agents.greedy, chronic, chronic, actions))
        out['greedy (simulate)'][-1]['simulations'] = agents.simulations
    return out


def evaluate_q(env, q: np.ndarray, actions: list, chronics=CHRONICS) -> list[dict]:
    """The greedy policy of the Q table on the evaluation chronics."""
    return [run_episode(env, QPolicy(q), chronic, chronic, actions) for chronic in chronics]

The agent's advantage is that it does not need those simulations at run time; its risk is that it cannot tell when its table is wrong.       def do_nothing(env: GridEnv, loading: np.ndarray) -> int:
    """Never acts."""
    return 0


def random_agent(env: GridEnv, loading: np.ndarray) -> int:
    """Acts at random when a line is above 90 %, as a baseline for the value of information."""
    return int(env.rng.integers(env.n_actions)) if loading.max() > 0.9 else 0


def greedy_agent(env: GridEnv, loading: np.ndarray, threshold: float = 0.9) -> int:
    """Simulates every action when the loading passes ``threshold`` and plays the one with the lowest maximum."""
    if loading.max() <= threshold:
        return 0
    worst = [env.simulate(a).max() for a in range(env.n_actions)]
    return int(np.argmin(worst))


def state_index(loading: np.ndarray, topology: int) -> int:
    """Discrete state: bin of the maximum loading, index of the most loaded line, current topology."""
    bin_index = int(np.searchsorted(RHO_BINS, loading.max()))
    return (bin_index * len(LINES) + int(loading.argmax())) * len(TOPOLOGIES) + topology


N_STATES = (len(RHO_BINS) + 1) * len(LINES) * len(TOPOLOGIES)


def run_episode(env: GridEnv, policy) -> dict:
    """Survival steps, mean maximum loading, total reward and wall-clock seconds per step of one episode."""
    loading, _ = env.reset()
    total, maxima, done = 0.0, [], False
    t0 = time.perf_counter()
    while not done:
        loading, reward, done = env.step(policy(env, loading))
        total += reward
        maxima.append(loading.max())
    return {
        'survival': env.step_index,
        'mean_max_rho': float(np.mean(maxima)),
        'reward': total,
        'ms_per_step': 1e3 * (time.perf_counter() - t0) / len(maxima),
    }


def frequent_actions(n_top: int = 2, seeds: range = range(100, 110)) -> list[int]:
    """Lab step 3: do-nothing plus the ``n_top`` actions the greedy agent played most often on training days."""
    counts = np.zeros(GridEnv.n_actions, dtype=int)
    for seed in seeds:
        env = GridEnv(seed)
        loading, _ = env.reset()
        done = False
        while not done:
            a = greedy_agent(env, loading)
            counts[a] += 1
            loading, _, done = env.step(a)
    counts[0] = 0
    return [0, *[int(a) for a in np.argsort(-counts)[:n_top] if counts[a] > 0]]


def q_learning(
    actions: list[int], episodes: int = 500, seed: int = 100, alpha: float = 0.2, gamma: float = 0.95, eps: float = 0.3
) -> np.ndarray:
    """Tabular Q-learning over the reduced action list on days drawn from ``seed``; returns the Q table.

    Exploration is epsilon-greedy with epsilon decaying linearly from ``eps`` to 0.01: with a constant epsilon the
    agent switches somewhere during the morning on almost every training day and never sees the meshed grid at
    the peak, so the states that matter are never learned.
    """
    rng = np.random.default_rng(seed)
    q = np.zeros((N_STATES, len(actions)))
    for episode in range(episodes):
        epsilon = max(0.01, eps * (1 - episode / episodes))
        env = GridEnv(int(rng.integers(1 << 31)))
        loading, topology = env.reset()
        s, done = state_index(loading, topology), False
        while not done:
            i = int(rng.integers(len(actions))) if rng.random() < epsilon else int(np.argmax(q[s]))
            loading, reward, done = env.step(actions[i])
            s_next = state_index(loading, env.topology)
            tripped = done and env.step_index < N_STEPS - 1
            target = reward if tripped else reward + gamma * q[s_next].max()
            q[s, i] += alpha * (target - q[s, i])
            s = s_next
    return q


def q_policy(q: np.ndarray, actions: list[int]):
    """The greedy policy of a Q table over the reduced action list, in the signature of the other agents."""

    def policy(env: GridEnv, loading: np.ndarray) -> int:
        return actions[int(np.argmax(q[state_index(loading, env.topology)]))]

    return policy

        super().__init__()

        self.base_seed, self.episode = seed, 0

        return np.append(np.minimum(obs.rho, 10.0), float(self.in_reference)).astype(np.float32)

    def reset(self, *, seed=None, options=None):
        """Start the next chronic (or ``options['chronic']``) at its first step, one day long."""
        super().reset(seed=seed)
        chronic = (
            options['chronic']
            if options and 'chronic' in options
            else self.chronics[self.episode % len(self.chronics)]
        )
        self.episode += 1
        self.in_reference, self.step_count = True, 0
        obs = lab.reset_day(self.env, int(chronic), self.base_seed + self.episode)
        self.last_obs = obs
        return self._observe(obs), {}

    def step(self, action: int):
        """Grid2Op step with the lab's reward; a game over before the end of the day terminates the episode."""
        obs, _, done, info = self.env.step(self.actions[int(action)])
        self.step_count += 1
        changed = int(action) != 0 and not info['is_illegal']
        self.in_reference = self.in_reference and not changed
        reward = lab.shaped_reward(obs, changed, done, self.step_count)
        self.last_obs = obs
        terminated = done and self.step_count < lab.N_STEPS
        truncated = done and not terminated
        return self._observe(obs), reward, terminated, truncated, info

      def init_params(n_in: int, n_out: int, hidden: int, key) -> dict:
    """Two dense layers with tanh in between, Glorot-style scaling."""
    k1, k2 = jax.random.split(key)
    return {
        'w1': jax.random.normal(k1, (n_in, hidden)) * jnp.sqrt(2 / (n_in + hidden)),
        'b1': jnp.zeros(hidden),
        'w2': jax.random.normal(k2, (hidden, n_out)) * jnp.sqrt(2 / (hidden + n_out)),
        'b2': jnp.zeros(n_out),
    }


def predict(params: dict, x: jnp.ndarray) -> jnp.ndarray:
    """Normalized injections in, normalized flows out."""
    return jnp.tanh(x @ params['w1'] + params['b1']) @ params['w2'] + params['b2']


def train(
    x: np.ndarray, y: np.ndarray, steps: int = STEPS, hidden: int = HIDDEN, seed: int = 0, lr: float = 3e-3
) -> dict:
    """Full-batch Adam on the mean squared error of the normalized targets; returns params and the scalers."""
    x_mean, x_std = x.mean(axis=0), x.std(axis=0) + 1e-9
    y_mean, y_std = y.mean(axis=0), y.std(axis=0) + 1e-9
    xn, yn = jnp.asarray((x - x_mean) / x_std), jnp.asarray((y - y_mean) / y_std)
    params = init_params(x.shape[1], y.shape[1], hidden, jax.random.PRNGKey(seed))
    opt = {k: (jnp.zeros_like(v), jnp.zeros_like(v)) for k, v in params.items()}

    def loss(p):
        return jnp.mean((predict(p, xn) - yn) ** 2)

    @jax.jit
    def step(p, m_v, t):
        g = jax.grad(loss)(p)
        new_p, new_mv = {}, {}
        for k in p:
            m = 0.9 * m_v[k][0] + 0.1 * g[k]
            v = 0.999 * m_v[k][1] + 0.001 * g[k] ** 2
            m_hat, v_hat = m / (1 - 0.9**t), v / (1 - 0.999**t)
            new_p[k] = p[k] - lr * m_hat / (jnp.sqrt(v_hat) + 1e-8)
            new_mv[k] = (m, v)
        return new_p, new_mv

    for t in range(1, steps + 1):
        params, opt = step(params, opt, t)
    return {'params': params, 'x_mean': x_mean, 'x_std': x_std, 'y_mean': y_mean, 'y_std': y_std}

    return np.asarray(predict(model['params'], jnp.asarray(xn))) * model['y_std'] + model['y_mean']

    Take the states \(s_0\) = "secure" and \(s_1\) = "overloaded", the actions \(a_0\) = wait and \(a_1\) = switch, and the model of value_iteration.py: waiting keeps a secure grid secure with probability 0.9 and an overloaded one overloaded with probability 0.7; switching lands in the secure state with probability 0.8 from either state; the rewards are \(R(s_0, a_0) = 0\), \(R(s_1, a_0) = -1\) (one unit per overloaded step) and \(R(s, a_1) = -0.1\) (the switching cost). The Bellman optimality equation is

    \[ V^*(s) = \max_{a} \Bigl[ R(s, a) + \gamma \sum_{s'} P(s' \mid s, a)\, V^*(s') \Bigr], \]

    that is, with \(\gamma = 0.9\),

    \[ \begin{aligned} V^*(s_0) &= \max\bigl\{\, 0.9\,[0.9\,V^*(s_0) + 0.1\,V^*(s_1)],\;\; -0.1 + 0.9\,[0.8\,V^*(s_0) + 0.2\,V^*(s_1)] \,\bigr\}, \\ V^*(s_1) &= \max\bigl\{\, -1 + 0.9\,[0.3\,V^*(s_0) + 0.7\,V^*(s_1)],\;\; -0.1 + 0.9\,[0.8\,V^*(s_0) + 0.2\,V^*(s_1)] \,\bigr\}. \end{aligned} \]

    Value iteration from \(V_0 = (0, 0)\): \(V_1 = (\max\{0, -0.1\}, \max\{-1, -0.1\}) = (0, -0.1)\); \(V_2 = (\max\{0.9 \cdot 0.1 \cdot (-0.1),\; -0.1 + 0.9 \cdot 0.2 \cdot (-0.1)\}, \max\{-1 + 0.9 \cdot 0.7 \cdot (-0.1),\; -0.1 + 0.9 \cdot 0.2 \cdot (-0.1)\}) = (-0.009, -0.118)\); \(V_3 = (-0.0179, -0.1277)\), and so on. The operator is a \(\gamma\)-contraction, so the error shrinks by at least 0.9 per step and the iteration converges, after 177 steps for a tolerance of \(10^{-10}\), to \(V^* = (-0.0989, -0.2088)\) with the policy "wait when secure, switch when overloaded". The check: solving \((I - \gamma P_\pi) V = R_\pi\) for that policy gives the same values, and the three other deterministic policies are worse: always wait \((-1.96, -4.13)\), always switch \((-1, -1)\), switch when secure and wait when overloaded \((-3.95, -5.58)\). The lesson for topology control is the structure of the answer: a small switching cost is worth paying as soon as the grid is overloaded, and not before.

    value_iteration.py
    def bellman_update(v: np.ndarray, gamma: float = GAMMA) -> tuple[np.ndarray, np.ndarray]:
    """One Bellman optimality step: Q(s, a) = R(a, s) + gamma sum_s' P(a, s, s') V(s'); returns max and argmax."""
    q = R + gamma * P @ v  # shape (action, state)
    return q.max(axis=0), q.argmax(axis=0)


def value_iteration(
    gamma: float = GAMMA, tol: float = 1e-10, max_iter: int = 10_000
) -> tuple[np.ndarray, np.ndarray, list]:
    """Iterate from V = 0 until the largest change is below ``tol``; returns V*, the greedy policy and the history."""
    v = np.zeros(len(R[0]))
    history = [v.copy()]
    for _ in range(max_iter):
        v_new, policy = bellman_update(v, gamma)
        history.append(v_new.copy())
        if np.abs(v_new - v).max() < tol:
            return v_new, policy, history
        v = v_new
    raise RuntimeError('value iteration did not converge')


def policy_value(policy: np.ndarray, gamma: float = GAMMA) -> np.ndarray:
    """Exact value of a deterministic policy: solve (I - gamma P_pi) V = R_pi."""
    p_pi = np.array([P[a, s] for s, a in enumerate(policy)])
    r_pi = np.array([R[a, s] for s, a in enumerate(policy)])
    return np.linalg.solve(np.eye(len(policy)) - gamma * p_pi, r_pi)

    action_space.py
    def assignments(elements: int, busbars: int = 2) -> int:
    """Every assignment of ``elements`` to labelled ``busbars``."""
    return busbars**elements


def assignments_up_to_symmetry(elements: int) -> int:
    """Two busbars whose swap gives the same topology: half of the labelled assignments."""
    return 2 ** (elements - 1)


def toop_splits(elements: int) -> int:
    """ToOp's valid splits of a substation with ``elements`` branches: both busbars keep at least two branches.

    Choosing the branches of the second busbar with 2 <= |B| <= d - 2 gives 2^d - 2 - 2d labelled assignments, each
    split counted twice (B and its complement), so 2^(d-1) - d - 1 splits; the unsplit configuration is one more.
    """
    return 2 ** (elements - 1) - elements - 1


def unitary_topologies(elements: int) -> int:
    """Grid2Op's ``get_all_unitary_topologies_set`` for one substation, counted by enumeration.

    Every assignment up to the busbar swap in which no busbar holds exactly one element, the unsplit one included;
    equals ``toop_splits + 1`` because both rules drop the same configurations.
    """
    configurations = set()
    for code in range(2**elements):
        second = frozenset(i for i in range(elements) if code >> i & 1)
        if len(second) in (1, elements - 1):
            continue
        configurations.add(frozenset([frozenset(range(elements)) - second, second]))
    return len(configurations)


def joint(per_substation: int, substations: int = 14) -> int:
    """Joint action space of independent choices at every substation."""
    return per_substation**substations

    flows_b(p, b) rebuilds \(B_f = \mathrm{diag}(b) A\) and \(B_{bus} = A^\top B_f\) from the susceptance vector, so jax.jacfwd(flows_b, argnums=1) is the \(3 \times 3\) matrix \(\partial f_\ell / \partial b_k\). For the injections of the worked example (\(p = (-1, 0.5)\) at buses 1 and 2, flows \((0.6176, -0.3824, -0.1176)\) on lines 0–1, 1–2, 0–2):

    \[ \frac{\partial f}{\partial b} = \begin{pmatrix} 0.0145 & -0.0360 & 0.0043 \\ 0.0145 & -0.0360 & 0.0043 \\ -0.0145 & 0.0360 & -0.0043 \end{pmatrix}. \]

    The central difference \((f(b + h e_2) - f(b - h e_2)) / 2h\) with \(h = 10^{-6}\) reproduces the third column to \(2 \times 10^{-11}\). Two structural facts are visible. The rows of lines 0–1 and 1–2 are equal and the row of 0–2 is their negative: in a single loop a change of any susceptance only moves flow between the two paths from bus 0 to bus 2 (through bus 1, or direct), so the nodal balances are untouched, \(A^\top\, \partial f / \partial b_k = 0\). And \(\sum_k b_k\, \partial f_\ell / \partial b_k = 0\): the flows are homogeneous of degree zero in \(b\), scaling every reactance by the same factor changes nothing. For an operator the gradient says how the flows move when a line's reactance changes: a series reactor or a phase-shifting transformer on line 1–2 (making \(b_1\) smaller) pushes flow from the path through bus 1 to the direct line at 0.036 MW per unit of susceptance, and the sign tells which way. It is the continuous relaxation of switching: \(b_k \to 0\) opens the line, and the derivative at the current \(b\) is the first-order estimate of the effect, exact for the linear DC model only along the path, since the map \(b \mapsto f\) is nonlinear (a rational function). It does not give the gradient with respect to a binary status, and the linear extrapolation to \(b_k = 0\) is wrong when the line carries much flow, which is why ToOp evaluates discrete topologies exactly (LODF, BSDF) instead of following this gradient.

    def flows_b(p_noslack: jnp.ndarray, b: jnp.ndarray) -> jnp.ndarray:
    """Branch flows for the injections ``p_noslack`` when the lines have susceptances ``b``."""
    b_f = b[:, None] * A
    b_bus = A.T @ b_f
    theta = jnp.concatenate([jnp.zeros(1), jnp.linalg.solve(b_bus[1:, 1:], p_noslack)])
    return b_f @ theta


def dflows_db(p_noslack: jnp.ndarray = P, b: jnp.ndarray = B) -> jnp.ndarray:
    """Jacobian of the flows with respect to the susceptances, shape (line, line), forward mode."""
    return jax.jacfwd(flows_b, argnums=1)(p_noslack, b)


def finite_difference_column(k: int, h: float = 1e-6, p_noslack: jnp.ndarray = P, b: jnp.ndarray = B) -> np.ndarray:
    """Central difference of the flows with respect to ``b[k]``: one column of the Jacobian."""
    e = jnp.zeros(len(b)).at[k].set(h)
    return np.asarray((flows_b(p_noslack, b + e) - flows_b(p_noslack, b - e)) / (2 * h))

    The GA is the Chapter 24 one: genomes of at most \(K = 2\) switched lines of IEEE 30 (loads scaled by 1.5 as in Chapter 27, exact fitness ch27_ieee30.overload_energy, islanding penalized), population 20, tournament of two, crossover by sampling from the union of the parents, mutation of one slot, elitist survival, 400 exact evaluations per run. The target is the exact optimum with two lines, 44.6 MW (lines 24–25 and 6–28, Chapter 27). The assisted variant generates five times as many offspring, scores them with a linear surrogate (one weight per line plus a bias, least squares on every genome evaluated so far) and evaluates exactly only the best 20 %. Over ten seeds:

    GA runs reaching 44.7 MW exact evaluations to target (median, min, max) best value, median
    plain 6 of 10 75 (10, 313) 44.6 MW
    assisted 3 of 10 23 (10, 25) 155.8 MW

    The surrogate does what it is supposed to when it is right: the three successful assisted runs reach the target in 23 exact evaluations against a median of 75. But it fails in seven runs out of ten, and those runs end far from the optimum (median best value 155.8 MW): the linear model assumes that switching effects are additive, and Chapter 24's Exercise 3 shows they are not (the pair 24–25 + 6–28 is much better than the sum of the two single switches would suggest). The surrogate therefore ranks the offspring by the single-line effects it has seen, steers the exact budget to combinations of individually good lines, and the population converges prematurely. This is the model-management problem of surrogate-assisted evolution (Jin 2011): a pre-screening model needs either a form that can represent interactions (pairwise terms, a Gaussian process, a GNN) or a fraction of the exact budget spent on offspring the surrogate ranks low, to keep correcting it. ToOp's two-stage design is the same idea with a physics model instead of a learned one: the DC solver is the cheap screen, exact for its own objective, and the AC stage the expensive evaluation; a learned surrogate would sit in front of the DC stage and inherit exactly this failure mode. A sign test on the ten paired runs gives no significant difference in evaluations-to-target because the assisted GA has only three successes; the honest summary is "faster when it works, and it works less often".

    def features(genome: np.ndarray, n_line: int) -> np.ndarray:
    """One-hot of the switched lines plus a bias: the linear surrogate assumes additive switching effects."""
    f = np.zeros(n_line + 1)
    f[genome[genome < n_line]] = 1
    f[-1] = 1
    return f


def exact(case: dict, genome: np.ndarray, n_line: int) -> float:
    """N-1 overload energy of the switched lines of a genome, islanding capped at ``PENALTY``."""
    return min(ch27_ieee30.overload_energy(case, tuple(int(g) for g in genome if g < n_line)), PENALTY)


def random_genome(rng: np.random.Generator, n_line: int) -> np.ndarray:
    """One to ``K_MAX`` distinct lines, sorted and padded with the sentinel ``n_line``."""
    k = rng.integers(1, K_MAX + 1)
    g = np.full(K_MAX, n_line)
    g[:k] = rng.choice(n_line, k, replace=False)
    return np.sort(g)


def offspring(rng: np.random.Generator, parents: np.ndarray, n_line: int) -> np.ndarray:
    """Tournament of two, crossover by sampling from the union of the parents, mutation of one slot."""
    a, b = parents[rng.integers(len(parents))], parents[rng.integers(len(parents))]
    pool = np.unique(np.concatenate([a[a < n_line], b[b < n_line]]))
    k = min(len(pool), rng.integers(1, K_MAX + 1)) if len(pool) else 0
    g = np.full(K_MAX, n_line)
    if k:
        g[:k] = rng.choice(pool, k, replace=False)
    if rng.random() < 0.5:
        g[rng.integers(K_MAX)] = n_line if rng.random() < 0.3 else rng.integers(n_line)
    g = np.unique(g[g < n_line])
    return np.concatenate([g, np.full(K_MAX - len(g), n_line)])


def run(case: dict, seed: int, assisted: bool, budget: int = BUDGET, pop: int = POP) -> dict:
    """One GA run: best value, exact evaluations spent and the evaluation at which ``TARGET`` was first reached."""
    rng = np.random.default_rng(seed)
    n_line = len(case['lines'])
    genomes = np.array([random_genome(rng, n_line) for _ in range(pop)])
    values = np.array([exact(case, g, n_line) for g in genomes])
    archive_x, archive_y = [features(g, n_line) for g in genomes], list(values)
    used, reached = pop, None
    best = values.min()
    if best <= TARGET:
        reached = int(np.argmax(values <= TARGET)) + 1
    while used < budget:
        order = np.argsort(values)
        parents = genomes[order[: pop // 2]]
        if assisted:
            candidates = np.array([offspring(rng, parents, n_line) for _ in range(5 * pop)])
            w, *_ = np.linalg.lstsq(np.array(archive_x), np.array(archive_y), rcond=None)
            scores = np.array([features(g, n_line) for g in candidates]) @ w
            children = candidates[np.argsort(scores)[:pop]]
        else:
            children = np.array([offspring(rng, parents, n_line) for _ in range(pop)])
        child_values = np.array([exact(case, g, n_line) for g in children])
        for g, v in zip(children, child_values, strict=True):
            used += 1
            archive_x.append(features(g, n_line))
            archive_y.append(v)
            if reached is None and v <= TARGET:
                reached = used
        genomes = np.vstack([genomes, children])
        values = np.concatenate([values, child_values])
        keep = np.argsort(values)[:pop]
        genomes, values = genomes[keep], values[keep]
        best = min(best, values.min())
    return {'best': float(best), 'evaluations': used, 'reached_at': reached}


def compare(case: dict, seeds: range = range(10)) -> dict[str, list[dict]]:
    """Plain against assisted GA on the same seeds."""
    return {'plain': [run(case, s, False) for s in seeds], 'assisted': [run(case, s, True) for s in seeds]}

    Exercise 5¶

    Brief: should the TSO deploy a reinforcement-learning topology agent?

    Recommendation. Not as a controller. Deploy it, if at all, as a proposal generator whose every suggestion is simulated before a human sees it, and measure it against the exhaustive one-step search and the rule base for a year before giving it more.

    What it offers. A trained policy answers in milliseconds without simulating hundreds of actions (Lab step 2: the greedy search costs one power flow per candidate action per step; the agent costs a table lookup or a forward pass), and the L2RPN results show agents that keep case-14 and case-118 grids alive through days the do-nothing baseline does not survive.

    Distribution shift. The policy is only as good as the days it has seen. Training uses historical or synthetic chronics; the next heat wave, a new wind farm, a different maintenance pattern or a market change produces states outside that distribution, and a table or network gives a confident answer there with no warning (Lab step 3: the agent cannot tell when its table is wrong). Retraining lags the grid; the lag is where the risk lives.

    Rare contingencies. Reinforcement learning optimizes expected return. An N-1 event that happens once a year is a negligible part of the return unless the reward is shaped to make it dominant, and then the agent learns a different, more conservative behaviour that costs every day. ToOp's overload_energy_n_1 is an N-1 metric by construction; an RL reward that is not will leave contingency security to luck.

    Guarantees. None: no feasibility guarantee (the proposed topology may island the grid, as four of the eight actions of the small environment do), no optimality, no bound on how far the policy is from the best action, and no explanation. A MILP gives bounds, ToOp gives a validated metric per proposal, the agent gives a number.

    Simulation checks. The non-negotiable element. Every proposal is run through obs.simulate or an AC N-1 analysis before it reaches the operator, and rejected when it does not improve the validated metrics. This turns the agent into a candidate generator, which is a role it can fill well, and removes the only failure mode that matters, a wrong action executed. The cost is one or a few simulations per step, which the agent was supposed to save; the saving is then the difference between simulating hundreds of actions and simulating one.

    How ToOp validates the proposals. ToOp's AC stage is exactly this check: the agent's topology is applied to the grid model, the AC N-1 is run, and evaluate_acceptance compares non-convergence, overload energy and critical branches with the unsplit grid (Chapter 28; mind its zero-baseline trap). An agent's proposal can also seed ToOp's DC repertoire, where it competes with the evolutionary candidates and is kept only if its cell has nothing better; the operator then reads one orao_summary.json whatever the source of the topology.

    Interpretability and the human in the loop. Operators act on proposals they can explain to the next shift. Present the agent's output as "topology X, validated: overload −40 %, no new critical branch, 3 switch operations", never as "the agent says X". Keep the rule base as the sanity check and the exhaustive search as the reference the agent must beat on the same seeds (Chapter 26) before any scope is widened.
